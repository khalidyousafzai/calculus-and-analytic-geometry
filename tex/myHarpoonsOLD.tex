\newcommand{\tvindexformat}[2]{%
    \setbox0=\hbox{$#1#2$}% Store the content box to get its exact width
    \text{\tikz[baseline=(content.base)]{
        % 1. Render the text as a named node
        \node[inner sep=0pt, outer sep=0pt] (content) {$#1#2$};
        % 2. Draw the harpoon matching the text width precisely
        \draw[
            -{Stealth[harpoon, length=3pt, width=2.5pt]}, % Adjust tip size here
            line width=0.4pt,                             % Arrow shaft thickness
        ] ([xshift=1.5pt, yshift=1.5pt]content.north west) -- ([xshift=-0.25pt, yshift=1.5pt]content.north east);
    }}%
}
