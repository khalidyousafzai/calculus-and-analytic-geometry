\باب{مسئلہ یولر   اور مسئلہ بڑھوتری}\شناخت{ضمیمہ_ط}
اس ضمیمہ میں مسئلہ یولر (حصہ \حوالہء{12.3}، مسئلہ \حوالہء{2})   اور دو متغیرات کا مسئلہ بڑھوتری(حصہ \حوالہء{12.4}، مسئلہ \حوالہء{3}) اخذ کیے جائیں  گے۔یولر نے \سن{1734} میں مسئلہ بیان کیا۔

\ابتدا{مسئلہءء}\موٹا{مسئلہ یولر}\\
اگر \عددی{f(x,y)} اور اس کے جزوی تفرق  \عددی{f_x}، \عددی{f_y}،  \عددی{f_{xy}}   اور \عددی{f_{yx}} ایک ایسے   پورے کھلے  خطے  میں معین ہوں جس میں نقطہ \عددی{(a,b)} پایا جاتا ہو اور   \عددی{(a,b)} پر   \عددی{f}،   \عددی{f_x}، \عددی{f_y}،  \عددی{f_{xy}}   اور \عددی{f_{yx}}استمراری ہوں تب \عددی{f_{xy}(a,b)=f_{yx}(a,b)} ہو گا۔
\انتہا{مسئلہءء}

\ابتدا{ثبوت}
 اوسط  قیمت مسئلے  (حصہ \حوالہء{3.2}،  مسئلہ  \حوالہء{4}) کو چار مرتبہ لاگو کر کے  \عددی{f_{xy}(a,b)} اور \عددی{f_{yx}(a,b)} کی  برابری  ثابت کی جا سکتی ہے۔  فرض کریں     مستوی \عددی{xy} میں مستطیل \عددی{R} کے اندرون میں  نقطہ \عددی{(a,b)} پایا جاتا ہے اور   مستطیل  \عددی{R} میں  \عددی{f}، \عددی{f_x}، \عددی{f_y}، \عددی{f_{xy}}، اور \عددی{f_{yx}}      معین ہیں۔ ہم ایسے اعداد \عددی{h} اور \عددی{k} لیتے ہیں کہ نقطہ  \عددی{(a+h,b+k)} بھی مستطیل \عددی{R} میں پایا جاتا ہو، اور درج ذیل فرق پر غور کرتے ہیں
\begin{align}\label{مساوات_ضمیمہ_یولر_الف}
\Delta=F(a+h)-F(a)
\end{align}
جہاں درج ذیل مراد ہے۔
\begin{align}\label{مساوات_ضمیمہ_یولر_ب}
F(x)=f(x,b+k)-f(x,b)
\end{align}
ہم اوسط قیمت مسئلے کا اطلاق \عددی{F} پر کرتے ہیں (جو قابل تفرق ہونے کی وجہ سے استمراری ہے)۔یوں مساوات \حوالہ{مساوات_ضمیمہ_یولر_الف} درج ذیل روپ اختیار کرتی ہے
\begin{align}\label{مساوات_ضمیمہ_یولر_پ}
\Delta=hF'(c_1)
\end{align}
جہاں \عددی{c_1}  اعداد  \عددی{a} اور  \عددی{a+h} کے بیچ ہو گا۔ مساوات \حوالہ{مساوات_ضمیمہ_یولر_ب} سے 
\begin{align*}
F'(x)=f_x(x,b+k)-f_x(x,b)
\end{align*}
لکھا جا سکتا ہے لہٰذا  مساوات \حوالہ{مساوات_ضمیمہ_یولر_پ} درج ذیل روپ اختیار کرتی ہے۔
\begin{align}\label{مساوات_ضمیمہ_یولر_ت}
\Delta=h[f_x(c_1,b+k)-f_x(c_1,b)]
\end{align}
اب ہم اوسط قیمت مسئلے کا اطلاق \عددی{g(y)=f_x(c_1,y)} پر کر کے 
\begin{align*}
g(b+k)-g(b)=kg'(d_1)
\end{align*}
یا
\begin{align*}
f_x(c_1,b+k)-f_x(c_1,b)=kf_{xy}(c_1, d_1)
\end{align*}
حاصل کرتے ہیں جہاں \عددی{d_1}  اعداد \عددی{b} اور \عددی{b+k} کے بیچ  ہو گا۔اس کو مساوات \حوالہ{مساوات_ضمیمہ_یولر_ت} میں ڈال  کر درج ذیل حاصل ہو گا
\begin{align}\label{مساوات_ضمیمہ_یولر_ٹ}
\Delta=hkf_{xy}(c_1,d_1)
\end{align}
جہاں  \عددی{(c_1,d_1)} مستطیل \عددی{R'} میں، جس کے راس \عددی{(a,b)}، \عددی{(a+h,b)}، \عددی{(a+h,b+k)}، اور \عددی{(a,b+k)} ہیں،    کوئی نقطہ ہے (شکل \حوالہء{A13} دیکھیں)۔

مساوات \حوالہ{مساوات_ضمیمہ_یولر_ب} سے مساوات \حوالہ{مساوات_ضمیمہ_یولر_الف} میں پُر  کرنے سے ذیل حاصل ہو گا
\begin{gather}
\begin{aligned}\label{مساوات_ضمیمہ_یولر_ث}
\Delta&=f(a+h,b+k)-f(a+h,b)-f(a,b+k)+f(a,b)\\
&=[f(a+h,b+k)-f(a,b+k)]-[f(a+h,b)-f(a,b)]\\
&=\phi(b+k)-\phi(b)
\end{aligned}
\end{gather}
جہاں \عددی{\phi} سے مراد   ذیل  ہے۔
\begin{align}\label{مساوات_ضمیمہ_یولر_ج}
\phi(y)=f(a+h,y)-f(a,y)
\end{align}
اوسط قیمت مسئلے کا اطلاق مساوات \حوالہ{مساوات_ضمیمہ_یولر_ث} پر کر کے ذیل حاصل ہو گا
\begin{align}\label{مساوات_ضمیمہ_یولر_چ}
\Delta=k\phi'(d_2)
\end{align}
جہاں \عددی{d_2} اعداد \عددی{b} اور \عددی{b+k} کے بیچ ہو گا۔ مساوات \حوالہ{مساوات_ضمیمہ_یولر_ج} سے ذیل ہو گا۔
\begin{align}\label{مساوات_ضمیمہ_یولر_ح}
\phi'(y)=f_y(a+h,y)-f_y(a,y)
\end{align}
مساوات \حوالہ{مساوات_ضمیمہ_یولر_ح} سے مساوات \حوالہ{مساوات_ضمیمہ_یولر_چ} میں پُر  کر کے ذیل ملتا ہے۔
\begin{align*}
\Delta=k[f_y(a+h,d_2)-f_y(a,d_2)]
\end{align*}
آخر میں ہم اوسط  قیمت  مسئلے کا اطلاق   قوسین میں بند  عبارت  پر کر کے ذیل حاصل کرتے ہیں
\begin{align}\label{مساوات_ضمیمہ_یولر_خ}
\Delta=hkf_{yx}(c_2,d_2)
\end{align}
جہاں \عددی{c_2} اعداد \عددی{a} اور \عددی{a+h} کے بیچ ہو گا۔

مساوات \حوالہ{مساوات_ضمیمہ_یولر_ٹ} اور مساوات \حوالہ{مساوات_ضمیمہ_یولر_خ} مل کر درج ذیل دیتی  ہیں
\begin{align}\label{مساوات_ضمیمہ_یولر_د} 
f_{xy}(c_1,d_1)=f_{yx}(c_2,d_2)
\end{align}
جہاں \عددی{(c_1,d_1)} اور \عددی{(c_2,d_2)} دونوں مستطیل \عددی{R'}   (شکل \حوالہء{A13}) میں پائے جاتے ہیں۔ مساوات \حوالہ{مساوات_ضمیمہ_یولر_د}  کہتی ہے کہ \عددی{f_{xy}} کی \عددی{(c_1,d_1)} پر قیمت \عددی{f_{yx}} کی \عددی{(c_2,d_2)} پر قیمت کے برابر ہے۔اگرچہ  بظاہر یہ وہ نتیجہ نہیں جو ہم چاہتے ہیں  تاہم اعداد \عددی{h} اور \عددی{k} کو ہم جتنا چاہیں چھوٹا کر  سکتے ہیں۔ چونکہ   \عددی{(a,b)}  پر \عددی{f_{xy}} اور \عددی{f_{yx}}  دونوں استمراری ہیں  لہٰذا \عددی{{f_{xy}(c_1,d_1)=f_{xy}(a,b)+\epsilon_1}}  اور 
\عددی{{f_{yx}(c_2,d_2)=f_{yx}(a,b)+\epsilon_2} } ہوں گے جہاں \عددی{h\to 0} اور \عددی{k\to 0} کرنے سے \عددی{\epsilon_1\to 0} اور \عددی{\epsilon_2\to 0} ہو گا۔ یوں \عددی{h\to 0} اور \عددی{k\to 0} کرنے سے  \عددی{f_{xy}(a,b)=f_{yx}(a,b)} ہو گا۔
\انتہا{ثبوت}

ہم \عددی{f_{xy}(a,b)} اور \عددی{f_{yx}(a,b)} کی برابری   کا ثبوت کمزور قیاس    کرتے ہوئے   دے سکتے ہیں۔  مثال کے طور پر، اتنا کافی ہے کہ \عددی{f}، \عددی{f_x}، اور  \عددی{f_y} مستطیل \عددی{R}  میں موجود ہوں اور \عددی{(a,b)} پر \عددی{f_{xy}} استمراری ہو۔ایسی صورت میں \عددی{(a,b)} پر \عددی{f_{yx}} موجود ہو گا اور اس نقطے پر \عددی{f_{xy}} کے برابر ہو گا۔

\ابتدا{مسئلہءء}\موٹا{دو متغیرات  کے تفاعلات کا مسئلہ بڑھوتری}\\
فرض کریں پورے  کھلے خطہ \عددی{R} میں، جس میں نقطہ \عددی{(x_0,y_0)} پایا جاتا ہے، \عددی{z=f(x,y)} کے یک رتبی جزوی تفرق معین ہیں، اور نقطہ \عددی{(x_0,y_0)} پر \عددی{f_x} اور \عددی{f_y} استمراری ہیں۔ ایسی صورت میں \عددی{R} میں رہ کر نقطہ
 \عددی{(x_0,y_0)} سے \عددی{(x_0+\Delta x,y_0+\Delta y)} منتقل ہونے پر \عددی{f} کی قیمت میں تبدیلی \عددی{\Delta z=f(x_0+\Delta x,y_0+\Delta y)-f(x_0,y_0)} درج ذیل روپ کی مساوات مطمئن کرتی ہے
\begin{align*}
\Delta z=f_x(x_0,y_0)\Delta x+f_y(x_0,y_0)\Delta y+\epsilon_1\Delta x+\epsilon_2\Delta y
\end{align*}
جہاں \عددی{\Delta x\to 0}، \عددی{\Delta y\to 0} کرنے سے \عددی{\epsilon_1\to 0}، \عددی{\epsilon_2\to 0}  ہو گا۔
\انتہا{مسئلہءء}

\ابتدا{ثبوت}
ہم \عددی{R} کے   اندر مستطیل \عددی{T} ، جس کا مرکز \عددی{A(x_0,y_0)} پر ہے،   استعمال کرتے ہیں، اور ہم فرض کرتے ہیں کہ \عددی{\Delta x} اور \عددی{\Delta y} کی لمبائیاں اتنی چھوٹی ہیں کہ \عددی{A} کو \عددی{B(x_0+\Delta x,y_0)} سے ملانے والا لکیری قطع اور \عددی{B} کو 
\عددی{C(x_0+\Delta x,y_0+\Delta y)} سے ملانے والا لکیری قطع \عددی{T} کے اندر پائے جاتے ہیں(شکل \حوالہء{A14})۔

ہم \عددی{\Delta z} کو دو بڑھوتریوں کا  مجموعہ \عددی{\Delta z=\Delta z_1+\Delta z_2} تصور کر سکتے ہیں جہاں  \عددی{A} سے \عددی{B} منتقل ہونے  پر  \عددی{f} کی قیمت میں تبدیلی درج ذیل ہے
\begin{align*}
\Delta z_1=f(x_0+\Delta x,y_0)-f(x_0,y_0)
\end{align*}
اور \عددی{B} سے \عددی{C} انتقال کے دوران  \عددی{f} کی قیمت میں تبدیلی درج ذیل ہے (شکل \حوالہء{A15})۔
\begin{align*}
\Delta z_2=f(x_0+\Delta x,y_0+\Delta y)-f(x_0+\Delta x,y_0)
\end{align*}
متغیر \عددی{x} کی  قیمتوں  کے اس بند وقفے پر  جو \عددی{x_0} کو \عددی{x_0+\Delta x} سے  ملاتی  ہیں،  \عددی{F(x)=f(x,x_0)} متغیر \عددی{x}  کا قابل تفرق (لہٰذا استمراری) تفاعل   ہو گا، اور اس  تفاعل کا تفرق درج ذیل ہو گا۔
\begin{align*}
F'(x)=f_x(x,y_0)
\end{align*}
اوسط قیمت مسئلے (حصہ \حوالہء{3.2}، مسئلہ \حوالہء{4}) کے تحت  \عددی{x_0} اور \عددی{x_0+\Delta x} کے بیچ  \عددی{x} کی ایک  قیمت \عددی{c}ایسی  ہو گی   جس پر  
\begin{align*}
F(x_0+\Delta x)-F(x_0)=F'(c)\Delta x
\end{align*}
یعنی
\begin{align*}
f(x_0+\Delta x,y_0)-f(x_0,y_0)=f_x(c,y_0)\Delta x
\end{align*}
یا
\begin{align}\label{مساوات_ضمیمہ_یولر_بارہ}
\Delta z_1=f_x(c,y_0)\Delta x
\end{align}
مطمئن ہو گا۔

اسی طرح \عددی{y_0} اور \عددی{y_0+\Delta y} کو ملانے والے \عددی{y} کے بند لکیری قطع  پر    \عددی{G(y)=f(x_0+\Delta x,y)}  متغیر \عددی{y} کا قابل تفرق (لہٰذا استمراری) تفاعل ہو گا، جس کا تفرق درج ذیل ہو گا۔
\begin{align*}
G'(y)=f_y(x_0+\Delta x,y)
\end{align*}
یوں \عددی{y_0} اور \عددی{y_0+\Delta y} کے بیچ  \عددی{y} کی ایک قیمت \عددی{d} ایسی ہو گی جس پر
\begin{align*}
G(y_0+\Delta y)-G(y_0)=G'(d)\Delta y
\end{align*}
یعنی
\begin{align*}
f(x_0+\Delta x,y_0+\Delta y)-f(x_0+\Delta x,y)=f_y(x_0+\Delta x,d)\Delta y
\end{align*}
یا
\begin{align}\label{مساوات_ضمیمہ_یولر_تیرہ}
\Delta z_2=f_y(x_0+\Delta x,d)\Delta y
\end{align}
مطمئن ہو گا۔

اب \عددی{\Delta x\to 0} اور \عددی{\Delta y\to 0}  کرنے سے \عددی{c\to x_0} اور \عددی{d\to y_0} ہو گا۔ یوں، چونکہ  \عددی{(x_0,y_0)} پر \عددی{f_x} اور \عددی{f_y} استمراری ہیں،لہٰذا  \عددی{\Delta x\to 0} اور \عددی{\Delta y\to 0} کرنے پر  درج ذیل دونوں مقادیر  صفر  کو پہنچتی ہیں۔
\begin{gather}
\begin{aligned}\label{مساوات_ضمیمہ_یولر_چودہ}
\epsilon_1&=f_x(c,y_0)-f_x(x_0,y_0)\\
\epsilon_2&=f_y(x_0+\Delta x,d)-f_y(x_0,y_0)
\end{aligned}
\end{gather}

آخر میں درج ذیل ہو گا
\begin{align*}
\Delta z&=\Delta z_1+\Delta z_2\\
&=f_x(c,y_0)\Delta x+f_y(x_0+\Delta x,d)\Delta y&&\text{\RL{مساوات \حوالہ{مساوات_ضمیمہ_یولر_بارہ} اور مساوات \حوالہ{مساوات_ضمیمہ_یولر_تیرہ} سے}}\\
&=[f_x(x_0,y_0)+\epsilon_1]\Delta x+[f_y(x_0,y_0)+\epsilon_2]\Delta y&&\text{\RL{مساوات \حوالہ{مساوات_ضمیمہ_یولر_چودہ} سے}}\\
&=f_x(x_0,y_0)\Delta x+f_y(x_0,y_0)\Delta y+\epsilon_1\Delta x+\epsilon_2\Delta y
\end{align*}
جہاں \عددی{\Delta x\to 0} اور \عددی{\Delta y\to 0} کرنے سے \عددی{\epsilon_1\to 0} اور \عددی{\epsilon_2\to 0} ہوں گے۔یوں ثبوت مکمل ہوا۔
\انتہا{ثبوت}

اسی طرح کے نتائج متناہی تعداد کے  غیر تابع متغیرات کے ہر تفاعل کے لئے مطمئن ہوتے ہیں۔ فرض کریں  کھلے خطہ پر، جس میں نقطہ  \عددی{(x_0,y_0,z_0)} پایا جاتا ہے،   درج ذیل کے یک رتبی جزوی تفرق معین ہیں،
\begin{align*}
w=f(x,y,z)
\end{align*}
اور \عددی{(x_0,y_0)} پر \عددی{f_x}، \عددی{f_y}، اور \عددی{f_z} استمراری ہیں۔ تب درج ذیل ہو گا
\begin{gather}
\begin{aligned}\label{مساوات_ضمیمہ_یولر_پندرہ}
\Delta w&=f(x_0+\Delta x,y_0+\Delta y,z_0+\Delta z)-f(x_0,y_0,z_0)\\
&=f_x\Delta x+f_y\Delta y +f_z\Delta z+\epsilon_1\Delta x+\epsilon_2\Delta y+\epsilon_3\Delta z
\end{aligned}
\end{gather}
جہاں  \عددی{\Delta x\to 0}، \عددی{\Delta y\to 0}، \عددی{\Delta z\to 0} کرنے سے \عددی{\epsilon_1\to 0}، \عددی{\epsilon_2\to 0}، \عددی{\epsilon_3\to 0} ہو گا۔

اس کلیہ میں تفرق \عددی{f_x}، \عددی{f_y}، \عددی{f_z} نقطہ \عددی{(x_0,y_0,z_0)} پر حاصل کیے جائیں گے۔

مساوات \حوالہ{مساوات_ضمیمہ_یولر_پندرہ} میں پیش نتیجے کا ثبوت، \عددی{\Delta w} کو  درج ذیل  تین بڑھوتریوں کا مجموعہ تصور کر کے
\begin{align}
\Delta w_1&=f(x_0+\Delta x,y_0,z_0)-f(x_0,y_0,z_0)\label{مساوات_ضمیمہ_بڑھوتری_ایک}\\
\Delta w_2&=f(x_0+\Delta x,y_0+\Delta y,z_0)-f(x_0+\Delta x,y_0,z_0)\label{مساوات_ضمیمہ_بڑھوتری_دو}\\
\Delta w_3&=f(x_0+\Delta x,y_0+\Delta y, z_0+\Delta z)-f(x_0+\Delta x,y_0+\Delta y,z_0)\label{مساوات_ضمیمہ_بڑھوتری_تین}
\end{align}
