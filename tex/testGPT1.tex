\section{معلمی سطحیں}

ہم نے مستوی میں منحنیات کو تین مختلف طریقوں سے بیان کیا ہے:

\begin{align*}
\text{صریح صورت:}\qquad & y=f(x) \\
\text{ضمنی صورت:}\qquad & F(x,y)=0 \\
\text{معلمی سمتی صورت:}\qquad & \mathbf{r}(t)=f(t)\mathbf{i}+g(t)\mathbf{j},
\qquad a\le t\le b
\end{align*}

فضا میں سطحوں کے لیے بھی اسی طرح کی تعریفیں موجود ہیں:

\begin{align*}
\text{صریح صورت:}\qquad & z=f(x,y) \\
\text{ضمنی صورت:}\qquad & F(x,y,z)=0
\end{align*}

ایک معلمی صورت بھی موجود ہے جس میں سطح پر کسی نقطے کی جگہ کو دو متغیرات کے سمتی تفاعل کی صورت میں ظاہر کیا جاتا ہے۔ اس حصہ میں ہم سطحی رقبہ اور سطحی تکملات کے مطالعہ کو ان سطحوں تک وسیع کریں گے جنہیں معلمی صورت میں بیان کیا گیا ہو۔

\subsection*{سطحوں کی معلمی نمائندگی}

فرض کریں

\begin{equation}
\mathbf{r}(u,v)
=
f(u,v)\mathbf{i}
+
g(u,v)\mathbf{j}
+
h(u,v)\mathbf{k}
\label{eq:16.6.1}
\end{equation}

ایک مسلسل سمتی تفاعل ہو جو \(uv\) مستوی کے ایک خطہ \(R\) پر معین ہو اور \(R\) کے داخلی حصے میں ایک ایک ہو (شکل \حوالہء{ 16.0})۔
ہم \(\mathbf{r}\) کی سعت کو \اصطلاح{سطح }\فرہنگ{سطح}\فرہنگ{surface}\(S\) کہتے ہیں جس کی تعریف یا نقشہ کشی \(\mathbf{r}\) کے ذریعہ کی جاتی ہے۔ مساوات \eqref{eq:16.6.1} اور  \(R\)  کا دائرہ کار مل کر اس سطح کی ایک  \اصطلاح{معلمی روپ}\فرہنگ{معلمی روپ}\حاشیہب{parametrization}\فرہنگ{parametrization} تشکیل دیتے ہیں۔ متغیرات \(u\) اور \(v\) \اصطلاح{معلمات}\فرہنگ{معلمات}\حاشیہب{parameters}\فرہنگ{parameters} کہلاتے ہیں جبکہ \(R\) کو \اصطلاح{معلمی دائرہ کار}\فرہنگ{معلمی 
دائرہ کار}\حاشیہب{parameter domain}\فرہنگ{parameter domain} کہا جاتا ہے۔

\begin{align*}

\end{align*}

\begin{align*}
\mathbf{r}(\phi,\theta)=&(a\sin\phi\cos\theta)\mathbf{i}+(a\sin\phi\sin\theta)\mathbf{j}+(a\cos\phi)\mathbf{k}\\
&0\le \phi \le \pi,\qquad0\le \theta \le 2\pi

اب \(\mathbf{r}_\phi \times \mathbf{r}_\theta\) نکالتے ہیں۔

\begin{align*}
\mathbf{r}_\phi \times \mathbf{r}_\theta&=
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
a\cos\phi\cos\theta & a\cos\phi\sin\theta & -a\sin\phi \\
-a\sin\phi\sin\theta & a\sin\phi\cos\theta & 0
\end{vmatrix}\\
&=(a^2\sin^2\phi\cos\theta)\mathbf{i}+(a^2\sin^2\phi\sin\theta)\mathbf{j}+(a^2\sin\phi\cos\phi)\mathbf{k}
\end{align*}

یوں درج ذیل ہو گا۔
\begin{align*}
\left|\mathbf{r}_\phi \times \mathbf{r}_\theta\right|&=\sqrt{a^4\sin^4\phi\cos^2\theta+a^4\sin^4\phi\sin^2\theta+a^4\sin^2\phi\cos^2\phi}\\
&=\sqrt{a^4\sin^4\phi+a^2\sin^2\phi\cos^2\phi}=\sqrt{a^4\sin^2\phi(\sin^2\phi+\cos^2\phi)}\\
&=a^2\sqrt{\sin^2\phi}=a^2\sin\phi.
\end{align*}
چونکہ \(0\le \phi \le \pi\) پر \(\sin\phi \ge 0\) ہو گا لہٰذا سطحی رقبہ درج ذیل ہو گا۔

A&=\int_{0}^{2\pi}\int_{0}^{\pi}a^2\sin\phi\,\dif \phi\,\dif\theta\\
&\int_{0}^{2\pi}\left[-a^2\cos\phi\right]_{0}^{\pi}\,\dif\theta=\int_{0}^{2\pi}2a^2\,\dif \theta=4\pi a^2\quad \text{\RL{مربع اکائیاں}}

\ابتدا{تعریف}\quad  سطحی تکمل برائے معلمی  سطح

اگر سطح  \عددی{S}  استمراری ہو  جس  کا معلمی روپ
\begin{align*}
\mathbf{r}(u,v) =& f(u,v)\mathbf{i} + g(u,v)\mathbf{j} + h(u,v)\mathbf{k}\\
&a \le u \le b,\; c \le v \le d
\end{align*}
ہو، اور  \(G(x,y,z)\)  ایک استمراری تفاعل ہو جو  \(S\) پر  معین  ہو، تب  \عددی{S} پر \عددی{G} کا تکمل درج ذیل ہو گا۔
\begin{align*}
\iint_S G(x,y,z)\, \dif \sigma=\int_c^d \int_a^bG\big(f(u,v), g(u,v), h(u,v)\big)\,\left| \mathbf{r}_u \times \mathbf{r}_v\right| \, \dif u\, \dif v
\end{align*}
\انتہا{تعریف}

\ابتدا{مثال}\quad معلمی روپ کی  سطح پر تکمل

تفاعل \عددی{G(x,y,z) = x^2} کا تکمل مخروطی سطح \عددی{z = \sqrt{x^2 + y^2},\, 0 \le z \le 1} پر معلوم کریں۔

حل:\quad
مثال \حوالہء{1} اور  مثال{4} کا  کام آگے بڑھاتے ہوئے \عددی{|\kvec{r}_r\times \kvec{r}_{\theta}|=\sqrt{2}r} لے کر درج ذیل ہو گا۔
\begin{align*}
\iint_S x^2 \, \dif \sigma&=\int_0^{2\pi} \int_0^1(r^2 \cos^2 u)(\sqrt{2}\, r)\, \dif r\, \dif u&& x=r\cos\theta\\
&=\sqrt{2}\int_0^{2\pi} \int_0^1r^3 \cos^2 \theta \, \dif r\, \dif \theta\\
&=\frac{\sqrt{2}}{4}\int_0^{2\pi} \cos^2 \theta \, \dif \theta=\frac{\sqrt{2}}{4}\left[  \frac{\theta}{2}+\frac{1}{4}\sin 2\theta \right]_0^{2\pi}=\frac{\pi\sqrt{2}}{2}
\end{align*}
\انتہا{مثال}

\ابتدا{مثال}\quad
بہاؤ کی تلاش

قطع مکافی بیلن \عددی{y = x^2,\, 0 \le x \le 1,\, 0 \le z \le 4}  سے \عددی{\kvec{F}=yz\ai+x\aj-z^2\ak} کا  باہر رخ گزرتا  بہاؤ تلاش کریں (شکل \حوالہء{16.57})۔

حل:\quad
سطح پر  \عددی{x=x}، \عددی{y=x^2}، اور \عددی{z=z} ہیں لہٰذا معلمی روپ \عددی{
\kvec{r}(x,z)=x\ai+x^2\aj+z\ak
} ہو گا جہاں \عددی{
{0\le x\le1},\,{0\le z\le 4}
}ہے۔

مماسی سمتیات کا صلیبی ضرب درج ذیل ہو گا۔

\begin{align*}
\mathbf{r}_x \times \mathbf{r}_z =
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
1 & 2x & 0 \\
0 & 0 & 1
\end{vmatrix}
= 2x\,\mathbf{i} - \mathbf{j}
\end{align*}

سطح پر باہر رخ عمودی سمتیہ درج ذیل ہے۔

\begin{align*}
\mathbf{n}=\frac{\kvec{r}_x\times\kvec{r}_z}{|\kvec{r}_x\times\kvec{r}_z|}=
\frac{2x\,\mathbf{i} - \mathbf{j}}{\sqrt{4x^2 + 1}}
\end{align*}

سطح پر  \عددی{y=x^2} ہے لہٰذا اس پر سمتیہ میدان درج ذیل ہو گا۔
\begin{align*}
\mathbf{F} = yz\ai+x\aj-z^2\ak=x^2 z\,\mathbf{i} + x\,\mathbf{j} - z^2\,\mathbf{k}
\end{align*}
یوں 
\begin{align*}
\mathbf{F} \cdot \mathbf{n}&=\frac{1}{\sqrt{4x^2+1}}((x^2z)(2x)+(x)(-1)+(-z^2)(0))\\
&=\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\end{align*}
%====================================================

اور \عددی{\kvec{F}} کا  سطح  سے  باہر رخ گزرتا  بہاؤ درج ذیل ہو گا۔
\begin{align*}
\iint_S \mathbf{F} \cdot \mathbf{n}\, \dif \sigma &=\int_0^4 \int_0^1\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\left| \mathbf{r}_x \times \mathbf{r}_z \right| \dif x\, \dif z\\
&=\int_0^4 \int_0^1\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\sqrt{4x^2+1}\dif x\, \dif z\\
&=\int_0^4 \int_0^1 (2x^3 z - x)\, \dif x\, \dif z=\int_0^4\left[ \frac{x^4 z}{2} - \frac{x^2}{2} \right]_0^1 \dif z\\
&\int_0^4 \frac{z - 1}{2}\, \dif z=\left.\frac{1}{4} (z - 1)^2\right]_0^4\\
&=\frac{1}{4}(9)-\frac{1}{4}(1)=2
\end{align*}
\انتہا{مثال}

\ابتدا{مثال}\quad
مرکز کمیت کی تلاش

مستقل کثافت  \عددی{\delta}  کا باریک خول  مخروط \عددی{z=\sqrt{x^2+y^2}} سے مستویات  \عددی{z=1} اور \عددی{z=2}  کاٹتی ہیں (شکل \حوالہء{16.58})۔ اس خول کا مرکز کمیت تلاش کریں۔ 

حل:\quad

محور \عددی{z} کے گرد  سطح تشاکل ہے لہٰذا \عددی{\bar{x}=0} اور \عددی{\bar{y}=0}  ہو گا۔ ہمیں \عددی{\bar{z} = \frac{M_{xy}}{M}} تلاش کرنی ہے۔ مثال \حوالہء{1} اور مثال \حوالہء{4} کی طرح چلتے ہوئے 
\begin{align*}
\mathbf{r}(r,\theta) =r \cos \theta\, \mathbf{i}+r \sin \theta\, \mathbf{j}+r\, \mathbf{k},\quad1 \le r \le 2,\; 0 \le \theta \le 2\pi
\end{align*}
اور
\begin{align*}
\left| \mathbf{r}_r \times \mathbf{r}_{\theta} \right| = \sqrt{2}\, r
\end{align*}
ہو گا۔ یوں درج ذیل ہو گا۔

\begin{align*}
M &=
\iint_S \delta \, \dif \sigma=\int_0^{2\pi} \int_1^2 \delta  \sqrt{2}\, r\, \dif r\, \dif \theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\left[ \frac{r^2}{2} \right]_1^2 \dif\theta=\delta\sqrt{2}\int_0^{2\pi}\left(2-\frac{1}{2}\right)\dif\theta\\
&=\delta \sqrt{2}\left[\frac{3\theta}{2}\right]_0^{2\pi}=3\pi\delta\sqrt{2}
\end{align*}
%
\begin{align*}
M_{xy}&=\iint_S \delta z\, \dif \sigma=\int_0^{2\pi} \int_1^2\delta r \sqrt{2}\, r\, \dif r\, \dif\theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\int_1^2 r^2\dif r\dif\theta =\delta\sqrt{2}\int_0^{2\pi}\left[\frac{r^3}{3}\right]_1^2\dif\theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\frac{7}{3}\dif\theta=\frac{14}{3}\pi\delta\sqrt{2}
\end{align*}
%
\begin{align*}
\bar{z}=\frac{M_{xy}}{M}=\frac{14\pi\delta\sqrt{2}}{3(3\pi\delta\sqrt{2})}=\frac{14}{9}
\end{align*}
خول کا مرکز کمیت نقطہ \عددی{(0,0,\tfrac{14}{9})} پر پایا جاتا ہے۔
\انتہا{مثال}

\حصہ{مسئلہ سٹوکس  }

ہم نے  حصہ\حوالہء{16.4 } میں دیکھا، دو بعدی میدان \عددی{\mathbf{F} = M\mathbf{i} + N\mathbf{j}}  میں  نقطہ $(x,y)$ پر کثافت  دائری بہاؤ یا   گردش  کا جزو درج ذیل  غیر سمتی  مقدار  \عددی{\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}}  بیان کرتا ہے۔ تین  ابعاد  میں  مستوی میں نقطہ $P$ کے گرد دائری بہاؤ  کو  سمتیہ کے ذریعے بیان کیا جاتا ہے۔یہ سمتیہ اس مستوی  کے عمود ہو گا   جس میں دائری بہاؤ  ہو(شکل\حوالہء{16.59})، اور سمتیہ  کا رخ دائری بہاؤ   کے خط    کے ساتھ دائیں ہاتھ کا تعلق رکھے گا۔سمتیہ کی لمبائی  سیال کے گھومنے کی شرح ظاہر کرتی ہے، جو  \عددی{P}  کے لحاظ سے دائری بہاؤ کا مستوی  جھکانے سے  عموماً  تبدیل ہو گی۔ہم کیا دیکھتے ہیں کہ  جب دائری بہاؤ کا سمتی رفتاری میدان \عددی{\kvec{F}=M\ai+N\aj+P\ak}  ہو،  زیادہ سے زیادہ دائری بہاؤ کا سمتیہ درج ذیل \اصطلاح{گردشی سمتیہ} \فرہنگ{گردشی سمتیہ}\حاشیہب{curl vector}\فرہنگ{curl vector} ہو گا۔
\begin{align}
\mathbf{F}\text{گردش}=\left(\frac{\partial P}{\partial y} - \frac{\partial N}{\partial z}\right)\mathbf{i}+
\left(\frac{\partial M}{\partial z} - \frac{\partial P}{\partial x}\right)\mathbf{j}+
\left(\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}\right)\mathbf{k}\tag{1}
\end{align}

یہ معلومات ہمیں\اصطلاح{مسئلہ سٹوکس }\فرہنگ{سٹوکس!مسئلہ}\حاشیہب{Stokes’ Theorem}\فرہنگ{Stokes!theorem}  دیتا ہے، جو مسئلہ گرین کے دائری بہاؤ و گردشی روپ کی  تین بعدی   وسعت ہے۔

دھیان رہے کہ \عددی{(\mathbf{F}\text{گردش}) \cdot \mathbf{k}=\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}} اور \عددی{\mathbf{F} = M(x,y)\mathbf{i} + N(x,y)\mathbf{j}}    کی صورت میں حصہ\حوالہء{16.4 } میں پیش کی گئی   تعریف   بلا تضاد  ہیں۔


مساوات \حوالہء{1} میں گردش \عددی{\kvec{F}}   کا  کلیہ عموماً        درج ذیل علامتی عامل  کے ذریعے لکھا جاتا ہے۔

\begin{align}
\nabla =\mathbf{i}\frac{\partial}{\partial x}+\mathbf{j}\frac{\partial}{\partial y}+\mathbf{k}\frac{\partial}{\partial z}\tag{2}
\end{align}
%===================================
%L 90977

(علامت $\nabla$ کو  ہم "ڈیل\حاشیہب{del} "  پڑھتے ہیں۔)  یہ عامل استعمال کر کے  گردش \عددی{\kvec{F}} \فرہنگ{گردش!کلیہ}\حاشیہب{curl}\فرہنگ{curl!equation} کو  ہم \عددی{\nabla\times\kvec{F}} لکھتے ہیں۔

\begin{align*}
\nabla \times \mathbf{F}&=\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
\frac{\partial}{\partial x} &
\frac{\partial}{\partial y} &
\frac{\partial}{\partial z} \\
M & N & P
\end{vmatrix}\\
&=\left(\frac{\partial P}{\partial y} - \frac{\partial N}{\partial z}\right)\mathbf{i}+
\left(\frac{\partial M}{\partial z} - \frac{\partial P}{\partial x}\right)\mathbf{j}+
\left(\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}\right)\mathbf{k}\\
&=\kvec{F}\text{گردش}
\end{align*}
مختصراً درج ذیل لکھا جائے گا۔
\begin{align}
  \mathbf{F}\text{گردش}=\nabla \times \mathbf{F}\tag{3}
\end{align}
%====================================================

\ابتدا{مثال}\موٹا{گردش \عددی{\kvec{F}} کی تلاش}

میدان \عددی{\mathbf{F} = (x^2 - y)\mathbf{i} + 4z\mathbf{j} + x^2\mathbf{k}}   کی گردش تلاش کریں۔

\موٹا{حل:}

\begin{align*}
\kvec{F}\text{گردش}&=\nabla\times \kvec{F}\\
&=
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
\frac{\partial}{\partial x} &
\frac{\partial}{\partial y} &
\frac{\partial}{\partial z} \\
x^2 - y & 4z & x^2
\end{vmatrix}\\
&=\left(\frac{\partial}{\partial y}(x^2) - \frac{\partial }{\partial z}(4z)\right)\mathbf{i}-
\left(\frac{\partial }{\partial x}(x^2) - \frac{\partial }{\partial z}(x^2-y)\right)\mathbf{j}+
\left(\frac{\partial}{\partial x}(4z) - \frac{\partial}{\partial y}(x^2 - y)\right)\mathbf{k}\\
&=(0 - 4)\ai-(2x - 0)\aj+(0 +1)\ak\\
&=-4\mathbf{i} - 2x\mathbf{j} + \mathbf{k}
\end{align*}
\انتہا{مثال}


جیسا کہ ہم دیکھیں گے، عامل  $\nabla$ کے کئی دیگر استعمال پائے جاتے ہیں۔مثلاً  جب  غیر سمتی تفاعل \عددی{f(x,y,z)} پر لگایا جائے،  یہ\اصطلاح{ڈھلوان }\فرہنگ{ڈھلوان!کلیہ}\حاشیہب{gradient}\فرہنگ{gradient!equation} دیگا۔

\begin{align}
\nabla f=\frac{\partial f}{\partial x}\mathbf{i}+\frac{\partial f}{\partial y}\mathbf{j}+\frac{\partial f}{\partial z}\mathbf{k}
\end{align}
اسے ہم   "ڈھلوان \عددی{f}"یا "ڈیل \عددی{f}" کہیں\حاشیہب{grad f} گے\حاشیہب{del f}۔

\
\حصہء{مسئلہ سٹوکس}

\اصطلاح{مسئلہ سٹوکس }\فرہنگ{سٹوکس!مسئلہ}\حاشیہب{Stokes' Theorem} \فرہنگ{Stokes!theorem}  کہتا ہے  کہ  ایسی صورتوں میں جو عملاً عام طور پائی جاتی ہیں،    فضا میں    سمت بند سطح  کی سرحد پر،      سطح کے عمودی اکائی سمتیہ \عددی{\kvec{n}} لے لحاظ سے خلاف گھڑی رخ  چلتے ہوئے (شکل \حوالہء{16.60}) ، سمتیہ میدان کا دائری بہاؤ  اور   سطح پر میدان کی گردش کے عمودی جزو کا تکمل برابر ہوں گے۔
%=====================
%L 91049


\ابتدا{مسئلہ}\موٹا{مسئلہ سٹوکس}
    سمت بند سطح \عددی{S}   کی سرحد پر،      سطح کے عمودی اکائی سمتیہ \عددی{\kvec{n}} لے لحاظ سے خلاف گھڑی  چلتے ہوئے سمتیہ میدان  \عددی{\kvec{F}=M\ai+N\aj+P\ak} کا دائری بہاؤ   \عددی{S} پر \عددی{\nabla\times \kvec{F}\cdot\kvec{n}} کے تکمل کے برابر ہو گا۔
\begin{align}
 \underbrace{\ointctrclockwise_C \mathbf{F} \cdot \dif \mathbf{r}}_{\text{\RL{خلاف گھڑی دائری بہاؤ}}}=\underbrace{\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{n}\, \dif \sigma}_{\text{\RL{تکمل گردش}}}  \tag{4}
\end{align}
\انتہا{مسئلہ}

ہم مساوات \حوالہء{4}  کو دیکھ کر کہہ سکتے ہیں کہ   ایسی دو مختلف سمت  بند سطحیں  \عددی{S_1} اور \عددی{S_2}  جن کی سرحد  \عددی{C} ایک ہو، کے تکمل گردش برابر ہوں گے۔

\begin{align*}
\iint_{S_1} (\nabla \times \mathbf{F}) \cdot \mathbf{n}_1\, \dif \sigma=
\iint_{S_2} (\nabla \times \mathbf{F}) \cdot \mathbf{n}_2\, \dif \sigma
\end{align*}


 دونوں تکمل  گردش  اس لئے ایک دوسرے کے   برابر ہوں گے  کیونکہ دونوں  مساوات \حوالہء{4} کے  بائیں ہاتھ  کے برابر ہیں، بشرطیکہ اکائی  عمودی  سمتیات \عددی{\kvec{n}_1} اور \عددی{\kvec{n}_2}   سطوح کی درست سمت بندی کریں۔

مسئلہ سٹوکس میں موجود  تکملات کی   وجودیت یقینی بنانے کے لئے ضرور ی ہے کہ \عددی{\kvec{F}}، \عددی{C}، اور \عددی{S} پر  ریاضیاتی شرائط مسلط ہوں گے۔ درکار عمومی شرائط  کہتے ہیں کہ تمام تفاعل، سمتیہ میدان، اور ان کے تفرق استمراری ہوں۔

اگر  $C$ مستوی  \عددی{xy}  میں   موجود ، خلاف گھڑی سمت بند  منحنی   ہو، اور \عددی{R} مستوی \عددی{xy} میں ایک خطہ ہو جس کو \عددی{C} گھیرتی ہو، تب \عددی{\dif \sigma=\dif x\dif y}  اور درج ذیل ہو گا۔
\begin{align*}
(\nabla \times \mathbf{F}) \cdot \mathbf{n}=(\nabla \times \mathbf{F}) \cdot \mathbf{k}=\left(
\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}\right)
\end{align*}
ان حالات میں مساوات سٹوکس درج ذیل صورت اختیار  کرتی ہے
\begin{align*}
\ointctrclockwise_C \mathbf{F} \cdot \dif \mathbf{r}=\iint_R\left(\frac{\partial N}{\partial x}-\frac{\partial M}{\partial y}\right)\dif x\dif y
\end{align*}
جو مسئلہ گرین میں  موجود مساوات کی دائری بہاؤ و گردشی روپ ہے۔اسی طرح انہیں قدموں پر  الٹ  چلتے ہوئے  ہم مسئلہ گرین کے  دائری بہاؤ و گردشی  روپ کو دو بعدی میدان کے لئے  درج ذیل ڈیل علامتیت میں لکھ سکتے ہیں۔
\begin{align}
\ointctrclockwise_C \mathbf{F} \cdot \dif \mathbf{r}=\iint_R (\nabla \times \mathbf{F}) \cdot \mathbf{k}\, \dif A \tag{5}
\end{align}
شکل\حوالہء{16.61} ملاحظہ کریں۔


\ابتدا{مثال}\quad \موٹا{نصف کرہ کے لئے مساوات سٹوکس کی تصدیق}\\
نصف کرہ \عددی{S:\, x^2+y^2+z^2=9,\, z\ge 0}، اس  کو گھیرتا دائرہ \عددی{C:\, x^2+y^2=9,\,z=0}، اور میدان \عددی{\kvec{F}=y\ai-x\aj} کے لئے مساوات \حوالہء{4} کی قیمت تلاش کریں۔
%=====================
%L 91122

\موٹا{حل:}\quad
معلمی روپ \عددی{\kvec{r}(\theta)=(3\cos\theta)\ai+(3\sin\theta)\aj,\,0\le \theta\le 2\pi} استعمال کرتے ہوئے اوپر  سے نظارہ کرتے ہوئے \عددی{C} کے گرد خلاف گھڑی دائری بہاؤ تلاش کرتے ہیں۔
\begin{align*}
\dif \mathbf{r}&=-3\sin \theta\,\dif \theta\,\mathbf{i}+3\cos \theta\,\dif\theta\,\mathbf{j}\\
\mathbf{F} &= y\mathbf{i} - x\mathbf{j}=3\sin \theta\,\mathbf{i}-3\cos \theta\,\mathbf{j}\\
\mathbf{F} \cdot \dif \mathbf{r}&=-9\sin^2 \theta\,\dif\theta-9\cos^2 \theta\,\dif\theta=-9\dif\theta\\
\ointctrclockwise_C \mathbf{F} \cdot \dif \mathbf{r}&=\int_0^{2\pi} -9\,\dif \theta=-18\pi
\end{align*}

اب \عددی{\kvec{F}}   کے لئے تکمل گردش تلاش کرتے ہیں۔
\begin{align*}
\nabla \times \mathbf{F}&= \left(\frac{\partial P}{\partial y}-\frac{\partial N}{\partial z}\right)\ai+
 \left(\frac{\partial M}{\partial z}-\frac{\partial P}{\partial x}\right)\aj+
 \left(\frac{\partial N}{\partial x}-\frac{\partial M}{\partial y}\right)\ak\\
&=(0-0)\ai+(0-0)\aj+(-1-1)\ak=-2\ak\\
\kvec{n}&=\frac{x\ai+y\aj+z\ak}{\sqrt{x^2+y^2+z^2}}=\frac{x\ai+y\aj+z\ak}{3}&&\text{\RL{بیرونی اکائی سمتیہ}}\\
\dif \sigma&=\frac{3}{z}\dif A&&\text{\RL{حصہ \حوالہء{16.5} کی مثال \حوالہء{5} میں \عددی{a=3}}}\\
\nabla\times \kvec{F}\cdot\kvec{n}\dif\sigma&=-\frac{2z}{3}\frac{3}{z}\dif A=-2\dif A
\end{align*}
یوں درج ذیل ہو گا۔
\begin{align*}
\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{n}\, \dif\sigma=\iint_{x^2+y^2\le 9} -2\,\dif A=-18\pi
\end{align*}

اس طرح دائرے کی ہمراہ دائری بہاو  اور گردش کا تکمل نصف کرہ پر  برابر ہیں، جیسا کہ ہونا چاہیے تھا۔
\انتہا{مثال}

\ابتدا{مثال}\quad \موٹا{دائری بہاؤ کی تلاش}\\
مستوی \عددی{z=2} مخروط \عددی{z=\sqrt{x^2+y^2}} سے منحنی \عددی{C}  پر ملتا ہے۔ اس منحنی پر ، اوپر سے دیکھتے ہوئے خلاف گھڑی (شکل \حوالہء{16.62})، میدان \عددی{\mathbf{F} = (x^2 - y)\mathbf{i} + 4z\mathbf{j} + x^2\mathbf{k}}  کا ، دائری بہاؤ تلاش کریں۔

سطح: \quad
مسئلہ سٹوکس کے استعمال سے ہم دائری بہاؤ سطح پر تکمل سے حاصل کرتے ہیں۔ اوپر سے دیکھتے ہوئے \عددی{C} پر گھڑی خلاف چلنا   \ترچھا{اندرونی }عمود  \عددی{\kvec{n}}  لینے کے مترادف ہے  جس کا  \عددی{z}   جزو مثبت ہو گا۔

ہم مخروط کا  معلمی روپ لکھتے  ہیں۔

\begin{align*}
\mathbf{r}(r,\theta)=r\cos \theta\,\mathbf{i}+r\sin \theta\,\mathbf{j}+r\,\mathbf{k},\quad0 \le r \le 2,\; 0 \le \theta \le 2\pi
\end{align*}
اس طرح درج ذیل ہو گا۔
\begin{align*}
\kvec{n}&=\frac{\mathbf{r}_r \times \mathbf{r}_{\theta}}{|\mathbf{r}_r \times \mathbf{r}_{\theta}|}=
\frac{- r\cos \theta\,\mathbf{i}-r\sin \theta\,\mathbf{j}+r\,\mathbf{k}}{r\sqrt{2}}&&
\text{\RL{حصہ \حوالہء{16.6} مثال \حوالہء{4}}}\\
&=\frac{-\cos \theta\,\mathbf{i} - \sin \theta\,\mathbf{j} + \mathbf{k}}{\sqrt{2}}\\
\dif \sigma & = r\sqrt{2}\,\dif r\,\dif \theta&&\text{\RL{حصہ \حوالہء{16.6} مثال \حوالہء{4}}}\\
\nabla \times \mathbf{F}&=-4\mathbf{i} - 2x\mathbf{j} + \mathbf{k}&&\text{\RL{مثال \حوالہء{1}}}\\
&=-4\mathbf{i} - 2r\cos \theta\,\mathbf{j} + \mathbf{k}&&  x=r\cos\theta
\end{align*}

\begin{align*}

\end{align*}

مسئلہ سٹوکس کے مطابق   یہ دائری بہاؤ  ، \عددی{C} میں محیط  سطح \عددی{S}  سے گزرتا \عددی{\nabla \times\kvec{F}} کے بہاؤ  کے برابر ہو گا۔
\begin{align*}
\ointctrclockwise_C \mathbf{F}\cdot \dif \mathbf{r}=\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{n}\, \dif\delta
\end{align*}
 ہم  \عددی{\kvec{F}} کے دائرہ کار میں    اٹل  نقطہ $Q$، اور    اس نقطہ پر   مخصوص سمت $\mathbf{u}$ منتخب کرتے  ہیں۔  فرض کریں  $C$ ایک دائرہ ہے جس کا رداس \عددی{\rho} اور مرکز \عددی{Q} ہے اور جس کا مستوی \عددی{\kvec{u}} کو عمودی ہے۔اگر \عددی{Q} پر \عددی{\nabla\times\kvec{F}} استمراری ہو،   \عددی{C} میں محیط دائری قرص \عددی{S} پر \عددی{\nabla\times\kvec{F}} کے \عددی{\kvec{u}} ویں جزو کی اوسط قیمت \عددی{\rho\to 0} کی صورت میں \عددی{Q}  پر \عددی{\nabla\times\kvec{F}} کے \عددی{\kvec{u}} ویں جزو  کو پہنچے گی۔
\begin{align*}
(\nabla \times \mathbf{F} \cdot \mathbf{u})_Q=\lim_{\rho\to 0}\frac{1}{\pi \rho^2}\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{u}\, \dif\sigma
\end{align*}
آخری مساوات میں سطحی تکمل کی جگہ دائری بہاؤ لکھ کر درج ذیل حاصل ہو گا۔
\begin{align}
(\nabla \times \mathbf{F} \cdot \mathbf{u})_Q=\lim_{\rho \to 0}\frac{1}{\pi \rho^2}\ointctrclockwise_C \mathbf{F}\cdot \dif \mathbf{r}\tag{6}
\end{align}
مساوات \حوالہء{6}    کے بائیں ہاتھ کی قیمت اس صورت زیادہ سے زیادہ ہو گی جب \عددی{\kvec{u}} کا رخ $\mathbf{u}$، $\nabla \times \mathbf{F}$ کی سمت میں ہو۔چھوٹے $\rho$ کے لئے مساوات \حوالہء{6}  کے دائیں ہاتھ   کا حد تخمیناً درج ذیل ہو گا، جو  \عددی{C} کے ہمراہ  دائری بہاؤ تقسیمِ رقبہ قرص    (دائری بہاؤ کی کثافت)ہے۔
\begin{align*}
\frac{1}{\pi \rho^2}\ointctrclockwise_C \mathbf{F}\cdot \dif \mathbf{r}
\end{align*}
فرض کریں \عددی{Q} پر رداس \عددی{\rho} کی چپو چرخی رکھی جاتی ہے جس کا دھرا  \عددی{\kvec{u}}  کے ہمراہ ہے۔ سیال کا \عددی{C} کے ہمراہ دائری بہاؤ  چپو چرخی کے   گھومنے کی رفتار پر اثر انداز ہو گا۔ جب  دائری بہاؤ تکمل کی قیمت زیادہ سے زیادہ ہو اس صورت چپو چرخی تیز سے تیز گھومے گی؛ یوں  جب دھرا \عددی{\nabla\times\kvec{F}} کے رخ ہو، چپو چرخی تیز سے تیز گھومے گی (شکل \حوالہء{16.63})۔

\ابتدا{مثال}\quad\موٹا{کثافت دائری بہاؤ اور \عددی{\nabla\times\kvec{F}} کا تعلق}\\
اٹل کثافت کا سیال محور \عددی{z} کے گرد \عددی{\kvec{v}=\omega(-y\ai+x\aj)}  سمتی رفتار سے گھوم رہا ہے جہاں \عددی{\omega} ایک مثبت  مستقل جس کو گھومنے  کا\اصطلاح{زاویائی رفتار }\فرہنگ{زاویائی!رفتار}کہتے ہیں(شکل \حوالہء{16.64})۔  اگر \عددی{\kvec{F}=\kvec{v}} ہو،  \عددی{\nabla\times \kvec{F}} کیا ہو گا اور کثافت دائری بہاؤ کے ساتھ اس کا تعلق کیا ہو گا؟


حل:\quad
چونکہ \عددی{\kvec{F}=\kvec{v}=-\omega y\ai+\omega x\aj} ہے لہٰذا درج ذیل ہو گا۔

\begin{align*}
\nabla \times \mathbf{F}=
\nabla\times\kvec{F}&=\left(\frac{\partial P}{\partial y}-\frac{\partial N}{\partial z}\right)\ai+
\left(\frac{\partial M}{\partial z}-\frac{\partial P}{\partial x}\right)\aj+
\left(\frac{\partial N}{\partial x}-\frac{\partial M}{\partial y}\right)\ak\\
&=(0-0)\ai+(0-0)\aj+(\omega-(-\omega))\ak=2\omega\ak
\end{align*}

مسئلہ سٹوکس  کے مطابق،    رداس \عددی{\rho} کے دائرہ \عددی{C}   کے ہمراہ، جو مستوی مثلاً مستوی  \عددی{xy}     میں قرص \عددی{S}  کا احاطہ کرتا ہو، کے ہمراہ \عددی{\kvec{F}} کا دائری بہاؤ درج ذیل ہو گا۔

\begin{align*}
\ointctrclockwise_C \mathbf{F} \cdot \dif \mathbf{r}=\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{n}\, \dif \sigma=\iint_S 2\omega \ak\cdot\ak \dif x\dif y=(2\omega)(\pi \rho^2)
\end{align*}
یوں درج ذیل ہو گا جو \عددی{\kvec{u}=\ak} لیتے ہوئے مساوات \حوالہء{6}  کے عین مطابق ہے۔
\begin{align*}
(\nabla \times \kvec{F})\cdot \ak=2\omega=\frac{1}{\pi\rho^2}\ointctrclockwise_C \mathbf{F} \cdot \dif \mathbf{r}
\end{align*}

\ابتدا{مثال}\quad \موٹا{مسئلہ سٹوکس کا اطلاق}\\

\موٹا{حل:}\quad

سرحد \عددی{C} پر خلاف گھڑی  حرکت کے مترادف ہے۔ مسئلہ سٹوکس استعمال کرنے کے لئے درج ذیل حاصل کرنے ہیں۔

\frac{\partial}{\partial z} \\

\جزوحصہء{کثیر سطحی  سطح کے لئے مسئلہ سٹوکس کا ثبوت}
فرض کریں $S$ کثیر سطحی سطح ہے جو  محدود تعداد  کے  مسطح حصوں  پر مشتمل  ہے (مثلاً شکل\حوالہء{16.66})۔ ہم  \عددی{S} کے علیحدہ علیحدہ حصوں  پر مسئلہ گرین کا اطلاق  کرتے ہیں۔  مستویات  کی دو  قسمیں ہیں۔
\begin{enumerate}[1.]
\item 
وہ  جو   ہر طرف سے  دوسرے حصوں سے گھیرا ہو۔
\item 
وہ جن کا ایک یا ایک سے زیادہ اطراف    کسی دوسرے حصے سے متصل نہ ہو۔
\end{enumerate}

سطح  \عددی{S}   کی سرحد \عددی{\Delta}    قسم   \عددی{2} مستویات کے ان کناروں پر مشتمل ہے جو دوسری مستویات سے متصل نہیں۔شکل \حوالہء{16.66} میں مثلثات   \عددی{EAB}، \عددی{BCE}، اور \عددی{CDE} سطح \عددی{S}  کے کچھ حصے کو ظاہر کرتے ہیں جبکہ \عددی{ABCD} سرحد \عددی{\Delta} کے کچھ حصے کو ظاہر کرتا ہے۔ تینوں مثلثات  پر بالترتیب مسئلہ گرین کا اطلاق کر کے نتائج جمع کر کے درج ذیل حاصل ہو گا۔
\begin{align}
\left(\ointctrclockwise_{EAB}+\ointctrclockwise_{BCE}+\ointctrclockwise_{CDE}\right) F\cdot \dif \kvec{r}=
\left( \iint _{EAB}+\iint_{BCE}+\iint_{CDE}\right)\nabla \times \mathbf{F}\cdot \mathbf{n}\, \dif\sigma\tag{7}
\end{align}
چونکہ مساوات \حوالہء{7} میں  بائیں ہاتھ کے تین لکیری تکملات میں  اندرونی   (متصل  مثلثات کے مشترک)اضلاع  پر  لکیری تکملات  کے جوڑے ایک دوسرے کو کاٹتے ہیں لہٰذا  تینوں   مل کر  کنارہ \عددی{ABCDE} پر   واحد ایک  تکمل دیتے ہیں۔مثال کے طور پر  مثلث \عددی{ABE} کے ضلع  \عددی{BE} پر لکیری تکمل کی علامت \عددی{(\pm)} مثلث \عددی{EBC} کے اسی ضلع پر لکیری تکمل کی علامت  \عددی{(\mp)}کا الٹ ہو گی۔ضلع \عددی{CE} پر بھی ایسا ہو گا۔ یوں مساوات \حوالہء{7}   درج ذیل روپ اختیار کرتی ہے۔
\begin{align*}
\ointctrclockwise_{ABCDE} \mathbf{F}\cdot \dif \mathbf{r}=\iint_{ABCDE}\nabla \times \mathbf{F}\cdot \mathbf{n}\, \dif\sigma
\end{align*}
تمام اضلاع پر مسئلہ گرین کا اطلاق کر کے درج ذیل حاصل ہو گا جو  درحقیقت کثیر سطحی سطح \عددی{S} کا مسئلہ سٹوکس ہے۔ مزید عمومی  سطوح کے   لئے   ثبوت آپ کو   اعلٰی نصاب  میں  ملیں گے۔
\begin{align*}
\ointctrclockwise_{\Delta} \mathbf{F}\cdot \dif\mathbf{r}=\iint_{S}\nabla \times \mathbf{F}\cdot \mathbf{n}\, \dif\sigma
\end{align*}

\جزوحصہء{مسئلہ سٹوکس برائے سوراخ دار سطوح}
مسئلہ گرین کو وسعت  دینے  کے طرز پر مسئلہ سٹوکس کو  ایسی  سمت بند  سطح \عددی{S} تک وسعت دی جا سکتی ہے جس میں ایک یا ایک سے زیادہ سوراخ ہوں (شکل \حوالہء{16.67})۔سطح \عددی{S} پر \عددی{\nabla\times\kvec{F}}  کے عمودی جزو کا   سطح تکمل تمام سرحدی منحنیات پر \عددی{\kvec{F}} کے مماسی جزو کے لکیری تکملات کے مجموعہ کے برابر ہو گا، جہاں   \عددی{S} کی سمت بندی کو مد نظر  رکھ کر منحنیات پر چلنا ہو گا۔

\جزوجزوحصہء{ایک اہم تماثل}
ریاضیات اور طبیعی سائنس میں درج ذیل تماثل بار بار پیش آتا ہے۔
\begin{align}
\nabla\times \nabla f=\kvec{0} \tag{8}
\end{align}
یہ تماثل  جو ہر اس تفاعل \عددی{f(x,y,z)} کے لئے  کارآمد ہے جس کے دوہرے جزوی تفرق استمراری ہوں، کہتا ہے کہ تفاعل \عددی{f(x,y,z)} کے ڈھلوان کی گردش   صفر ہو گی۔اس کا ثبوت درج ذیل ہے۔
\begin{align*}
\nabla \times \nabla f =
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
\frac{\partial}{\partial x} & \frac{\partial}{\partial y} & \frac{\partial}{\partial z} \\
\frac{\partial f}{\partial x} & \frac{\partial f}{\partial y} & \frac{\partial f}{\partial z}
\end{vmatrix}
=(\, f_{zy} - f_{yz} \,) \mathbf{i}-(\, f_{zx} - f_{xz} \,) \mathbf{j}+(\, f_{yx} - f_{xy} \,) \mathbf{k}
\end{align*}
ثانی  جزوی تفرق استمراری ہونے کی صورت میں  قوسین میں بند  مختلف اقسام کے ثانی  تفرق  برابر ہوں گے   (حصہ \حوالہء{14.3}، مسئلہ \حوالہء{2}) لہٰذا سمتیہ صفر ہو گا

\حصہء{بقائی میدان اور مسئلہ سٹوکس}
ہم نے حصہ \حوالہء{16.3} میں دیکھا  کہ کھلے خطہ \عددی{D} میں میدان \عددی{\kvec{F}} کا بقائی ہونا اس کے مترادف ہے کہ  \عددی{D} میں ہر بند راہ پر \عددی{\kvec{F}} کا لکیری تکمل صفر ہو۔یہ از خود، \ترچھا{سادہ تعلق }خطوں میں، اس کے مترادف ہے کہ \عددی{\nabla\times\kvec{F}=\kvec{0}} ہو۔

\ابتدا{مسئلہ}\quad \موٹا{\عددی{\nabla\times \kvec{F}=\kvec{0}}  اور بند حلقہ  خاصیت کا تعلق} \\
اگر خلا میں سادہ تعلق آزاد خطہ \عددی{D} کے ہر نقطہ پر \عددی{\nabla\times\kvec{F}=\kvec{0}} ہو تب \عددی{D} میں  ٹکڑوں میں  ہموار بند راہ \عددی{C} پر درج ذیل ہو گا۔
\begin{align*}
\ointctrclockwise_C \kvec{F}\cdot\dif\kvec{r}=0
\end{align*}
\انتہا{مسئلہ}
\ابتدا{ثبوت}
\موٹا{خاکہ}\quad
مسئلہ \حوالہء{6} کا ثبوت عموماً دو مراحل میں کیا جاتا ہے۔ پہلا مرحلہ سادہ بند منحنیات کے لیے ہو گا۔ اعلٰی ریاضیات کی ایک شاخ \اصطلاح{ربڑی ہندسہ}\فرہنگ{ربڑی ہندسہ}\حاشیہب{topology}\فرہنگ{topology}   کا ایک مسئلہ  کہتا ہے کہ سادہ تعلق کھلے خطہ \عددی{D} میں ہر قابل تفرق   سادہ بند منحنی  \عددی{C}  ایک ہموار دو طرفہ سطح \عددی{S}،   جو \عددی{D} میں پائی جاتی ہو، کی سرحد ہو گی ۔یوں مسئلہ سٹوکس کے تحت درج ذیل ہو گا۔
\begin{align*}
\ointctrclockwise_C \mathbf{F} \cdot \dif\mathbf{r}=\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{n}\, \dif\sigma= 0
\end{align*}

دوسرا مرحلہ اُن منحنیات کے لیے ہے جو خود کو کاٹتی ہیں، جیسے کہ شکل\حوالہء{16.68 }میں دکھائی گئی ہے۔ ایسی منحنیات کو سادہ  حلقات، جن  کا احاطہ قابل سمت بند سطوح کرتی ہوں،  میں تقسیم کر کے باری باری ایک ایک حلقہ پر مسئلہ سٹوکس کا اطلاق کر کے نتائج کا مجموعہ لیا جاتا ہے۔
