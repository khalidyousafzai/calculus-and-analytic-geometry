%Q17
\ابتدا{سوال}\موٹا{گردش \عددی{G} کا پھیلاؤ صفر ہو گا}\\
\begin{enumerate}[a.]
\item
دکھائیں اگر  میدان \عددی{\kvec{G}=M\ai+N\aj+P\ak} کے مطلوبہ جزوی تفرق استمراری  ہوں، تب \عددی{\nabla\cdot \nabla\times \kvec{G}=0} ہو گا۔
\item
کیا آپ اس سے  ، میدان \عددی{\nabla\times \kvec{G}} کا بند سطح سے گزرتے بہاؤ کے بارے میں ،  کوئی   نتیجہ اخذ  کر  سکتے ہیں؟ اپنے جواب کی وجہ پیش کریں۔
\end{enumerate}
\انتہا{سوال}
%Q18
\ابتدا{سوال}
فرض کریں \عددی{\kvec{F}_1} اور \عددی{\kvec{F}_2}  قابل تفرق سمتی میدان ہیں جبکہ \عددی{a} اور \عددی{b}  حقیقی اختیاری مستقل ہیں۔ درج ذیل تماثل کی تصدیق کریں۔
\begin{enumerate}[a.]
\item
\(\nabla\cdot(a\kvec{F}_1+b\kvec{F}_2)=a\nabla\cdot\kvec{F}_1+b\nabla\cdot\kvec{F}_2\)
\item
\(\nabla\times(a\kvec{F}_1+b\kvec{F}_2)=a\nabla\times\kvec{F}_1+b\nabla\times\kvec{F}_2\)
\item
\(\nabla\cdot(\kvec{F}_1\times\kvec{F}_2)=\kvec{F}_2\cdot\nabla\times\kvec{F}_1-\kvec{F}_1\cdot\nabla\times\kvec{F}_2\)
\end{enumerate}
\انتہا{سوال}
%Q19
\ابتدا{سوال}
فرض کریں \عددی{\kvec{F}} قابل تفرق سمتی میدان  جبکہ  \عددی{g(x,y,z)} قابل تفرق غیر سمتی میدان ہے۔ درج ذیل تماثل کی تصدیق کریں۔
\begin{enumerate}[a.]
\item
\(\nabla\cdot(g\kvec{F})=g\nabla\cdot\kvec{F}+\nabla g\cdot \kvec{F}\)
\item
\(\nabla\times(g\kvec{F})=g\nabla\times\kvec{F}+\nabla g\times\kvec{F}\)
\end{enumerate}
\انتہا{سوال}

\ابتدا{سوال}
قابل تفرق سمتی میدان \عددی{\kvec{F}=M\ai+N\aj+P\ak} کے لئے علامتی اظہار \عددی{\kvec{F}\cdot\nabla}  کی تعریف درج ذیل ہے۔
\[M\frac{\partial}{\partial x}+N\frac{\partial}{\partial y}+P\frac{\partial}{\partial z}\]
قابل تفرق سمتی میدان \عددی{\kvec{F}_1} اور \عددی{\kvec{F}_2} کے لئے درج ذیل کی تصدیق کریں۔
\begin{enumerate}[a.]
\item
\(\nabla\times(\kvec{F}_1\times\kvec{F}_2)=(\kvec{F}_2\cdot\nabla)\kvec{F}_1-(\kvec{F}_1\cdot\nabla)\kvec{F}_2+(\nabla\cdot\kvec{F}_2)\kvec{F}_1-(\nabla\cdot\kvec{F}_1)\kvec{F}_2\)
\item
\(\nabla(\kvec{F}_1\cdot\kvec{F}_2)=(\kvec{F}_1\cdot\nabla)\kvec{F}_2+(\kvec{F}_2\cdot\nabla)\kvec{F}_1+\kvec{F}_1\times(\nabla\times\kvec{F}_2)+\kvec{F}_2\times(\nabla\times\kvec{F}_1)\)
\end{enumerate}
\انتہا{سوال}

\ابتدا{سوال}\موٹا{ہارمونی تفاعلات}\quad
فضا کے خطہ  \عددی{D} میں تفاعل \عددی{f(x,y,z)} اس صورت \اصطلاح{ہارمونی }\فرہنگ{ہارمونی}\حاشیہب{harmonic}\فرہنگ{harmonic}کہلاتا ہے  جب وہ پورے \عددی{D} میں درج ذیل  لاپلاسی مساوات کو مطمئن کرتا ہو۔
\[\nabla^2f=\nabla\cdot\nabla f=\frac{\partial^2f}{\partial x^2}+\frac{\partial^2f}{\partial y^2}+\frac{\partial^2f}{\partial z^2}=0\]
\begin{enumerate}[a.]
\item
فرض کریں ہموار سطح \عددی{S}    ، جس کا منتخب اکائی عمودی سمتیہ \عددی{\an} ہے، محدود خطہ  \عددی{D} کو گھیرتی ہے اور \عددی{f} 
اس  پورے خطے میں ہارمونی ہے۔ دکھائیں کہ \عددی{S} پر \عددی{\nabla f\cdot \an} کا تکمل،  یعنی \عددی{\an} کے رخ \عددی{f}  کا تفرق، صفر ہے۔
\item
دکھائیں اگر \عددی{D} میں \عددی{f} ہارمونی ہو تب درج ذیل ہو گا۔
\[\iint_Sf\nabla f\cdot \an\dif\sigma=\iiint_D|\nabla f|^2\dif H\]
\end{enumerate}
\انتہا{سوال}

\ابتدا{سوال}\شناخت{سوال_سمتی_میدان_گرین_کلیہ_اول}\موٹا{گرین کا کلیہ اول}\quad
ٹکڑوں میں ہموار بند  سطح \عددی{S} میں بند  پورے خطے \عددی{D}  پر غیر سمتی تفاعلات \عددی{f} اور \عددی{g}  کے یک رتبی اور  دو رتبی جزوی تفرقات استمراری ہیں۔دکھائیں کہ درج ذیل ہو گا۔
\begin{align}\label{مساوات_سمتی_میدان_گرین_کلیہ_اول}
\iint_S f\nabla g\cdot \an\dif\sigma=\iiint_D (f\nabla^2g+\nabla f\cdot \nabla g)\dif H
\end{align}
مساوات \حوالہ{مساوات_سمتی_میدان_گرین_کلیہ_اول} \اصطلاح{گرین کا کلیہ اول }\فرہنگ{گرین!کلیہ اول}\حاشیہب{Greens's first formula}\فرہنگ{Greens'!first formula}ہے۔ (اشارہ: میدان \عددی{\kvec{F}=f\nabla g} پر مسئلہ پھیلاؤ کا اطلاق کریں۔)
\انتہا{سوال}

\ابتدا{سوال}\موٹا{گرین کا کلیہ دوم}\quad (\ترچھا{سوال \حوالہ{سوال_سمتی_میدان_گرین_کلیہ_اول} کو آگے بڑھاتے ہیں۔})
مساوات \حوالہ{مساوات_سمتی_میدان_گرین_کلیہ_اول} میں \عددی{f} اور \عددی{g}  کی جگہیں آپس میں تبدیل کر کے اسی قسم کا دوسرا کلیہ  اخذ کریں۔اس کلیہ کو مساوات \حوالہ{مساوات_سمتی_میدان_گرین_کلیہ_اول} سے منفی کر کے درج ذیل دکھائیں۔
\begin{align}\label{مساوات_سمتی_میدان_گرین_کلیہ_دوم}
\iint_S (f\nabla g-g\nabla f)\cdot \an\dif\sigma=\iiint_D (f\nabla^2g-g\nabla^2f)\dif H
\end{align}
مساوات \حوالہ{مساوات_سمتی_میدان_گرین_کلیہ_دوم} \اصطلاح{گرین کا کلیہ دوم }\فرہنگ{گرین!کلیہ دوم}\حاشیہب{Green's second formula}\فرہنگ{Green's!second formula}ہے۔ 
\انتہا{سوال}

\ابتدا{سوال}\شناخت{سوال_سمتی_میدان_کمیت_بقا}\موٹا{کمیت کی بقا}\quad
فضا میں خطہ \عددی{D} پر \عددی{\kvec{v}(t,x,y,z)}  استمراری قابل تفرق سمتی میدان ہے، جبکہ \عددی{p(t,x,y,z)}  استمراری قابل تفرق غیر سمتی تفاعل ہے۔ متغیر \عددی{t} وقت کو ظاہر کرتا ہے۔ \اصطلاح{کمیت کی بقا کا قانون }\فرہنگ{قانون!بقا کمیت}\حاشیہب{Law of Conservation of Mass}\فرہنگ{law!conservation of mass}درج ذیل کہتا ہے:
\[\frac{\dif}{\dif t}\iiint_D p(t,x,y,z)\dif H=-\iint_S p\kvec{v}\cdot\an\dif\sigma\]
جہاں \عددی{S} وہ سطح ہے جو \عددی{D} کو گھیرتی ہے۔
\begin{enumerate}[a.]
\item
سمتی بہاؤ میدان  \عددی{\kvec{v}} اور لمحہ \عددی{t}  پر نقطہ \عددی{(x,y,z)} پر سیال کی کثافت \عددی{p} لے کر کمیت کی بقا کے قانون کا طبیعی مفہوم پیش کریں۔
\item
مسئلہ پھیلاؤ اور    لیبنٹز کا قاعدہ:
\[\frac{\dif}{\dif t}\iiint_D p(t,x,y,z)\dif H=\iiint_D \frac{\partial p}{\partial t}\dif H\]
\end{enumerate}
استعمال کر کے دکھائیں کہ کمیت کی بقا کا قانون استمراری مساوات:
\[\nabla \cdot p\kvec{v}+\frac{\partial p}{\partial t}=0\]
کا معادل  ہے۔(اول رکن  \عددی{\nabla \cdot p\kvec{v}} میں  متغیر \عددی{t} اٹل رکھا جاتا ہے، جبکہ دوسرے رکن \عددی{\partial p/\partial t} میں، ہم فرض کرتے ہیں کہ نقطہ \عددی{(x,y,z)} خطہ  \عددی{D} میں   اٹل رکھا گیا ہے۔)
\انتہا{سوال}
%Q32
\ابتدا{سوال}\موٹا{نفوذ حرّ کی مساوات}\quad
فرض کریں فضا میں خطہ \عددی{D}  میں مکین  ٹھوس جسم  کے نقطہ \عددی{(x,y,z)} کا درجہ حرارت لمحہ \عددی{t} پر  تفاعل \عددی{T(t,x,y,z)} دیتا ہو جس کے دوہرے تفرقات استمراری ہیں۔ جسم کی حری استعداد \عددی{c} اور کمیتی  کثافت \عددی{\rho} ہے۔مقدار \عددی{c\rho T} جسم کی  فی اکائی حجم حری توانائی کہلاتی ہے۔
\begin{enumerate}[a.]
\item
بتلائیں \عددی{-\nabla T} کیوں حرارت کے بہاؤ کا رخ دیگا۔
\item
\اصطلاح{بہاؤ  حر کا سمتیہ }\فرہنگ{بہاؤ حر سمتیہ}\حاشیہب{energy flux vector}\فرہنگ{energy flux vector} \عددی{-k\nabla T} لیں۔ (یہاں مستقل \عددی{k} \اصطلاح{موصلیت }\فرہنگ{موصلیت}\حاشیہب{conductivity}\فرہنگ{conductivity}کہلاتا ہے۔) کمیتی بقا کے قانون  (سوال \حوالہ{سوال_سمتی_میدان_کمیت_بقا}) میں \عددی{-k\nabla T=\kvec{v}} اور \عددی{c\rho T=p} لے کر مساوات نفوذ (برائے حرارت):
\[\frac{\partial T}{\partial t}=K\nabla^2 T\]
اخذ کریں، جہاں \عددی{K=k/(c\rho)>0}  \اصطلاح{نفوذیت }\فرہنگ{نفوذیت!مستقل}\حاشیہب{diffusivity}\فرہنگ{diffusivity!constant} کا مستقل کہلاتا ہے۔(دھیان رہے اگر یکساں موصل سلاخ جس کے سرے مکمل غیر موصل رکھے گئے ہوں کے نقطہ \عددی{x}  اور لمحہ \عددی{t} پر  درجہ حرارت کو  \عددی{T(t,x)}  ظاہر کرتا ہو ، تب \عددی{\nabla^2T=\partial^2T/\partial x^2} ہو گا اور مساوات نفوذ ، باب \حوالہ{باب_کثیر_المتغیر_تفاعل_اور_جزوی_تفرقات} کے  اضافی سوالات میں دی گئی یک بعدی مساوات حر دیگی۔)
\end{enumerate}
\انتہا{سوال}
