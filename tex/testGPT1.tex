\section{معلمی سطحیں}

ہم نے مستوی میں منحنیات کو تین مختلف طریقوں سے بیان کیا ہے:

\begin{align*}
\text{صریح صورت:}\qquad & y=f(x) \\
\text{ضمنی صورت:}\qquad & F(x,y)=0 \\
\text{معلمی سمتی صورت:}\qquad & \mathbf{r}(t)=f(t)\mathbf{i}+g(t)\mathbf{j},
\qquad a\le t\le b
\end{align*}

فضا میں سطحوں کے لیے بھی اسی طرح کی تعریفیں موجود ہیں:

\begin{align*}
\text{صریح صورت:}\qquad & z=f(x,y) \\
\text{ضمنی صورت:}\qquad & F(x,y,z)=0
\end{align*}

ایک معلمی صورت بھی موجود ہے جس میں سطح پر کسی نقطے کی جگہ کو دو متغیرات کے سمتی تفاعل کی صورت میں ظاہر کیا جاتا ہے۔ اس حصہ میں ہم سطحی رقبہ اور سطحی تکملات کے مطالعہ کو ان سطحوں تک وسیع کریں گے جنہیں معلمی صورت میں بیان کیا گیا ہو۔

\subsection*{سطحوں کی معلمی نمائندگی}

فرض کریں

\begin{equation}
\mathbf{r}(u,v)
=
f(u,v)\mathbf{i}
+
g(u,v)\mathbf{j}
+
h(u,v)\mathbf{k}
\label{eq:16.6.1}
\end{equation}

ایک مسلسل سمتی تفاعل ہو جو \(uv\) مستوی کے ایک خطہ \(R\) پر معین ہو اور \(R\) کے داخلی حصے میں ایک ایک ہو (شکل \حوالہء{ 16.0})۔
ہم \(\mathbf{r}\) کی سعت کو \اصطلاح{سطح }\فرہنگ{سطح}\فرہنگ{surface}\(S\) کہتے ہیں جس کی تعریف یا نقشہ کشی \(\mathbf{r}\) کے ذریعہ کی جاتی ہے۔ مساوات \eqref{eq:16.6.1} اور  \(R\)  کا دائرہ کار مل کر اس سطح کی ایک  \اصطلاح{معلمی روپ}\فرہنگ{معلمی روپ}\حاشیہب{parametrization}\فرہنگ{parametrization} تشکیل دیتے ہیں۔ متغیرات \(u\) اور \(v\) \اصطلاح{معلمات}\فرہنگ{معلمات}\حاشیہب{parameters}\فرہنگ{parameters} کہلاتے ہیں جبکہ \(R\) کو \اصطلاح{معلمی دائرہ کار}\فرہنگ{معلمی 
دائرہ کار}\حاشیہب{parameter domain}\فرہنگ{parameter domain} کہا جاتا ہے۔

\begin{align*}

\end{align*}

\begin{align*}
\mathbf{r}(\phi,\theta)=&(a\sin\phi\cos\theta)\mathbf{i}+(a\sin\phi\sin\theta)\mathbf{j}+(a\cos\phi)\mathbf{k}\\
&0\le \phi \le \pi,\qquad0\le \theta \le 2\pi

اب \(\mathbf{r}_\phi \times \mathbf{r}_\theta\) نکالتے ہیں۔

\begin{align*}
\mathbf{r}_\phi \times \mathbf{r}_\theta&=
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
a\cos\phi\cos\theta & a\cos\phi\sin\theta & -a\sin\phi \\
-a\sin\phi\sin\theta & a\sin\phi\cos\theta & 0
\end{vmatrix}\\
&=(a^2\sin^2\phi\cos\theta)\mathbf{i}+(a^2\sin^2\phi\sin\theta)\mathbf{j}+(a^2\sin\phi\cos\phi)\mathbf{k}
\end{align*}

یوں درج ذیل ہو گا۔
\begin{align*}
\left|\mathbf{r}_\phi \times \mathbf{r}_\theta\right|&=\sqrt{a^4\sin^4\phi\cos^2\theta+a^4\sin^4\phi\sin^2\theta+a^4\sin^2\phi\cos^2\phi}\\
&=\sqrt{a^4\sin^4\phi+a^2\sin^2\phi\cos^2\phi}=\sqrt{a^4\sin^2\phi(\sin^2\phi+\cos^2\phi)}\\
&=a^2\sqrt{\sin^2\phi}=a^2\sin\phi.
\end{align*}
چونکہ \(0\le \phi \le \pi\) پر \(\sin\phi \ge 0\) ہو گا لہٰذا سطحی رقبہ درج ذیل ہو گا۔

A&=\int_{0}^{2\pi}\int_{0}^{\pi}a^2\sin\phi\,\dif \phi\,\dif\theta\\
&\int_{0}^{2\pi}\left[-a^2\cos\phi\right]_{0}^{\pi}\,\dif\theta=\int_{0}^{2\pi}2a^2\,\dif \theta=4\pi a^2\quad \text{\RL{مربع اکائیاں}}

\ابتدا{تعریف}\quad  سطحی تکمل برائے معلمی  سطح

اگر سطح  \عددی{S}  استمراری ہو  جس  کا معلمی روپ
\begin{align*}
\mathbf{r}(u,v) =& f(u,v)\mathbf{i} + g(u,v)\mathbf{j} + h(u,v)\mathbf{k}\\
&a \le u \le b,\; c \le v \le d
\end{align*}
ہو، اور  \(G(x,y,z)\)  ایک استمراری تفاعل ہو جو  \(S\) پر  معین  ہو، تب  \عددی{S} پر \عددی{G} کا تکمل درج ذیل ہو گا۔
\begin{align*}
\iint_S G(x,y,z)\, \dif \sigma=\int_c^d \int_a^bG\big(f(u,v), g(u,v), h(u,v)\big)\,\left| \mathbf{r}_u \times \mathbf{r}_v\right| \, \dif u\, \dif v
\end{align*}
\انتہا{تعریف}

\ابتدا{مثال}\quad معلمی روپ کی  سطح پر تکمل

تفاعل \عددی{G(x,y,z) = x^2} کا تکمل مخروطی سطح \عددی{z = \sqrt{x^2 + y^2},\, 0 \le z \le 1} پر معلوم کریں۔

حل:\quad
مثال \حوالہء{1} اور  مثال{4} کا  کام آگے بڑھاتے ہوئے \عددی{|\kvec{r}_r\times \kvec{r}_{\theta}|=\sqrt{2}r} لے کر درج ذیل ہو گا۔
\begin{align*}
\iint_S x^2 \, \dif \sigma&=\int_0^{2\pi} \int_0^1(r^2 \cos^2 u)(\sqrt{2}\, r)\, \dif r\, \dif u&& x=r\cos\theta\\
&=\sqrt{2}\int_0^{2\pi} \int_0^1r^3 \cos^2 \theta \, \dif r\, \dif \theta\\
&=\frac{\sqrt{2}}{4}\int_0^{2\pi} \cos^2 \theta \, \dif \theta=\frac{\sqrt{2}}{4}\left[  \frac{\theta}{2}+\frac{1}{4}\sin 2\theta \right]_0^{2\pi}=\frac{\pi\sqrt{2}}{2}
\end{align*}
\انتہا{مثال}

\ابتدا{مثال}\quad
بہاؤ کی تلاش

قطع مکافی بیلن \عددی{y = x^2,\, 0 \le x \le 1,\, 0 \le z \le 4}  سے \عددی{\kvec{F}=yz\ai+x\aj-z^2\ak} کا  باہر رخ گزرتا  بہاؤ تلاش کریں (شکل \حوالہء{16.57})۔

حل:\quad
سطح پر  \عددی{x=x}، \عددی{y=x^2}، اور \عددی{z=z} ہیں لہٰذا معلمی روپ \عددی{
\kvec{r}(x,z)=x\ai+x^2\aj+z\ak
} ہو گا جہاں \عددی{
{0\le x\le1},\,{0\le z\le 4}
}ہے۔

مماسی سمتیات کا صلیبی ضرب درج ذیل ہو گا۔

\begin{align*}
\mathbf{r}_x \times \mathbf{r}_z =
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
1 & 2x & 0 \\
0 & 0 & 1
\end{vmatrix}
= 2x\,\mathbf{i} - \mathbf{j}
\end{align*}

سطح پر باہر رخ عمودی سمتیہ درج ذیل ہے۔

\begin{align*}
\mathbf{n}=\frac{\kvec{r}_x\times\kvec{r}_z}{|\kvec{r}_x\times\kvec{r}_z|}=
\frac{2x\,\mathbf{i} - \mathbf{j}}{\sqrt{4x^2 + 1}}
\end{align*}

سطح پر  \عددی{y=x^2} ہے لہٰذا اس پر سمتیہ میدان درج ذیل ہو گا۔
\begin{align*}
\mathbf{F} = yz\ai+x\aj-z^2\ak=x^2 z\,\mathbf{i} + x\,\mathbf{j} - z^2\,\mathbf{k}
\end{align*}
یوں 
\begin{align*}
\mathbf{F} \cdot \mathbf{n}&=\frac{1}{\sqrt{4x^2+1}}((x^2z)(2x)+(x)(-1)+(-z^2)(0))\\
&=\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\end{align*}
%====================================================

اور \عددی{\kvec{F}} کا  سطح  سے  باہر رخ گزرتا  بہاؤ درج ذیل ہو گا۔
\begin{align*}
\iint_S \mathbf{F} \cdot \mathbf{n}\, \dif \sigma &=\int_0^4 \int_0^1\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\left| \mathbf{r}_x \times \mathbf{r}_z \right| \dif x\, \dif z\\
&=\int_0^4 \int_0^1\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\sqrt{4x^2+1}\dif x\, \dif z\\
&=\int_0^4 \int_0^1 (2x^3 z - x)\, \dif x\, \dif z=\int_0^4\left[ \frac{x^4 z}{2} - \frac{x^2}{2} \right]_0^1 \dif z\\
&\int_0^4 \frac{z - 1}{2}\, \dif z=\left.\frac{1}{4} (z - 1)^2\right]_0^4\\
&=\frac{1}{4}(9)-\frac{1}{4}(1)=2
\end{align*}
\انتہا{مثال}

\ابتدا{مثال}\quad
مرکز کمیت کی تلاش

مستقل کثافت  \عددی{\delta}  کا باریک خول  مخروط \عددی{z=\sqrt{x^2+y^2}} سے مستویات  \عددی{z=1} اور \عددی{z=2}  کاٹتی ہیں (شکل \حوالہء{16.58})۔ اس خول کا مرکز کمیت تلاش کریں۔ 

حل:\quad

محور \عددی{z} کے گرد  سطح تشاکل ہے لہٰذا \عددی{\bar{x}=0} اور \عددی{\bar{y}=0}  ہو گا۔ ہمیں \عددی{\bar{z} = \frac{M_{xy}}{M}} تلاش کرنی ہے۔ مثال \حوالہء{1} اور مثال \حوالہء{4} کی طرح چلتے ہوئے 
\begin{align*}
\mathbf{r}(r,\theta) =r \cos \theta\, \mathbf{i}+r \sin \theta\, \mathbf{j}+r\, \mathbf{k},\quad1 \le r \le 2,\; 0 \le \theta \le 2\pi
\end{align*}
اور
\begin{align*}
\left| \mathbf{r}_r \times \mathbf{r}_{\theta} \right| = \sqrt{2}\, r
\end{align*}
ہو گا۔ یوں درج ذیل ہو گا۔

\begin{align*}
M &=
\iint_S \delta \, \dif \sigma=\int_0^{2\pi} \int_1^2 \delta  \sqrt{2}\, r\, \dif r\, \dif \theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\left[ \frac{r^2}{2} \right]_1^2 \dif\theta=\delta\sqrt{2}\int_0^{2\pi}\left(2-\frac{1}{2}\right)\dif\theta\\
&=\delta \sqrt{2}\left[\frac{3\theta}{2}\right]_0^{2\pi}=3\pi\delta\sqrt{2}
\end{align*}
%
\begin{align*}
M_{xy}&=\iint_S \delta z\, \dif \sigma=\int_0^{2\pi} \int_1^2\delta r \sqrt{2}\, r\, \dif r\, \dif\theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\int_1^2 r^2\dif r\dif\theta =\delta\sqrt{2}\int_0^{2\pi}\left[\frac{r^3}{3}\right]_1^2\dif\theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\frac{7}{3}\dif\theta=\frac{14}{3}\pi\delta\sqrt{2}
\end{align*}
%
\begin{align*}
\bar{z}=\frac{M_{xy}}{M}=\frac{14\pi\delta\sqrt{2}}{3(3\pi\delta\sqrt{2})}=\frac{14}{9}
\end{align*}
خول کا مرکز کمیت نقطہ \عددی{(0,0,\tfrac{14}{9})} پر پایا جاتا ہے۔
\انتہا{مثال}

\حصہ{سٹوکس  کا نظریہ }

جیسا کہ ہم نے  حصہ  16.4 میں دیکھا، دو بعدی میدان
\begin{align*}
\mathbf{F} = M\mathbf{i} + N\mathbf{j}
\end{align*}
میں کسی نقطہ $(x,y)$ پر گردش کی کثافت (circulation density) یا curl کا جزو درج ذیل  غیر سمتی  مقدار سے ظاہر ہوتا ہے:

\begin{align*}
\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}
\end{align*}

تین بعدی (three-dimensional) صورت میں کسی نقطہ $P$ کے گرد کسی سطح پر گردش کو ایک سمتیہ کے ذریعے بیان کیا جاتا ہے۔

یہ سمتیہ اس سطح کے عمود (عمودی) ہوتا ہے جس میں گردش ہو رہی ہو (شکل 16.59)، اور اس کی سمت دائیں ہاتھ کے اصول (right-hand rule) کے مطابق گردش کی سمت سے متعلق ہوتی ہے۔

اس سمتیہ کی لمبائی گردش کی شدت (rate of rotation) کو ظاہر کرتی ہے، جو عموماً اس بات پر منحصر ہوتی ہے کہ گردش والا طیارہ نقطہ $P$ کے گرد کیسے جھکا ہوا ہے۔

یہ ثابت کیا جا سکتا ہے کہ بہاؤ (flow) کے سمتیہ میدان
\begin{align*}
\mathbf{F} = M\mathbf{i} + N\mathbf{j} + P\mathbf{k}
\end{align*}
میں زیادہ سے زیادہ گردش (maximum circulation) کا سمتیہ درج ذیل ہوتا ہے جسے curl کہا جاتا ہے:

\begin{align*}
\nabla \times \mathbf{F}
=
\left(
\frac{\partial P}{\partial y} - \frac{\partial N}{\partial z}
\right)\mathbf{i}
+
\left(
\frac{\partial M}{\partial z} - \frac{\partial P}{\partial x}
\right)\mathbf{j}
+
\left(
\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}
\right)\mathbf{k}
\tag{1}
\end{align*}

یہ معلومات ہمیں سٹوکس کے نظریہ (Stokes’ Theorem) سے حاصل ہوتی ہیں، جو کہ Green’s Theorem کے circulation–curl روپ کی تین بعدی توسیع ہے۔

\vspace{1em}

یہ بات قابلِ غور ہے کہ:

\begin{align*}
(\nabla \times \mathbf{F}) \cdot \mathbf{k}
=
\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}
\end{align*}

یہی وہی تعریف ہے جو ہم نے  حصہ  16.4 میں دی تھی جب
\begin{align*}
\mathbf{F} = M(x,y)\mathbf{i} + N(x,y)\mathbf{j}
\end{align*}
تھا۔

curl کا یہ روپ   اکثر    ایک علامتی آپریٹر کے ذریعے بھی لکھا جاتا ہے:

\begin{align*}
\nabla =
\mathbf{i}\frac{\partial}{\partial x}
+
\mathbf{j}\frac{\partial}{\partial y}
+
\mathbf{k}\frac{\partial}{\partial z}
\tag{2}
\end{align*}

%L 90977

\vspace{1em}

\موٹا{(نوٹ):} علامت $\nabla$ کو "ڈیل" (del) پڑھا جاتا ہے۔

curl of $\mathbf{F}$ کو درج ذیل طور پر لکھا جاتا ہے:

\begin{align*}
\nabla \times \mathbf{F}
=
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
\frac{\partial}{\partial x} &
\frac{\partial}{\partial y} &
\frac{\partial}{\partial z} \\
M & N & P
\end{vmatrix}
\end{align*}

\begin{align*}
=
\left(
\frac{\partial P}{\partial y} - \frac{\partial N}{\partial z}
\right)\mathbf{i}
+
\left(
\frac{\partial M}{\partial z} - \frac{\partial P}{\partial x}
\right)\mathbf{j}
+
\left(
\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}
\right)\mathbf{k}
\end{align*}

اس کو مختصر طور پر یوں لکھا جاتا ہے:

\begin{align*}
\nabla \times \mathbf{F} = \text{curl } \mathbf{F}
\tag{3}
\end{align*}

\vspace{1em}

\begin{center}
{\Large مثال 1: curl معلوم کرنا}
\end{center}

میدان دیا گیا ہے:

\begin{align*}
\mathbf{F} = (x^2 - y)\mathbf{i} + 4z\mathbf{j} + x^2\mathbf{k}
\end{align*}

curl معلوم کریں۔

\موٹا{حل:}

\begin{align*}
\nabla \times \mathbf{F}
=
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
\frac{\partial}{\partial x} &
\frac{\partial}{\partial y} &
\frac{\partial}{\partial z} \\
x^2 - y & 4z & x^2
\end{vmatrix}
\end{align*}

\begin{align*}
=
\mathbf{i}\left(\frac{\partial x^2}{\partial y} - \frac{\partial 4z}{\partial z}\right)
-
\mathbf{j}\left(\frac{\partial x^2}{\partial x} - \frac{\partial x^2 - y}{\partial z}\right)
+
\mathbf{k}\left(\frac{\partial 4z}{\partial x} - \frac{\partial (x^2 - y)}{\partial y}\right)
\end{align*}

اب جزوی مشتقات:

\begin{align*}
=
\mathbf{i}(0 - 4)
-
\mathbf{j}(2x - 0)
+
\mathbf{k}(0 - (-1))
\end{align*}

\begin{align*}
=
-4\mathbf{i} - 2x\mathbf{j} + \mathbf{k}
\end{align*}

\vspace{1em}

جیسا کہ ہم دیکھیں گے، آپریٹر $\nabla$ کے مزید بھی استعمال ہیں۔

مثال کے طور پر جب اسے کسی scalar function $f(x,y,z)$ پر لگایا جائے تو یہ gradient دیتا ہے:

\begin{align*}
\nabla f
=
\frac{\partial f}{\partial x}\mathbf{i}
+
\frac{\partial f}{\partial y}\mathbf{j}
+
\frac{\partial f}{\partial z}\mathbf{k}
\end{align*}

اسے "grad f" یا "del f" بھی کہا جاتا ہے۔

\vspace{1em}

\begin{center}
{\Large سٹوکس کا نظریہ (Stokes’ Theorem)}
\end{center}

سٹوکس کا نظریہ کہتا ہے کہ عملی طور پر عام شرائط کے تحت:

کسی سمتیہ فیلڈ کی سطح کے کنارے (boundary curve) کے گرد گردش (circulation)،
اس سطح پر curl کے نارمل جزو کے سطحی تکمل کے برابر ہوتا ہے۔

یہ boundary curve $C$ کی سمت اس طرح منتخب کی جاتی ہے کہ وہ سطح کے unit عمودی vector $\mathbf{n}$ کے مطابق right-hand rule کو فالو کرے (شکل 16.60)۔

\vspace{1em}

\begin{center}
{\Large نظریہ 5: سٹوکس کا نظریہ (Stokes’ Theorem)}
\end{center}

سٹوکس کا نظریہ کہتا ہے کہ:

سمتیہ فیلڈ
\begin{align*}
\mathbf{F} = M\mathbf{i} + N\mathbf{j} + P\mathbf{k}
\end{align*}
کی گردش (circulation) کسی بند سرحدی منحنی $C$ کے گرد، جو کسی اورینٹڈ سطح $S$ کی boundary ہو، اس سطح پر curl کے نارمل جزو کے سطحی تکمل کے برابر ہوتی ہے، بشرطیکہ سمت (orientation) دائیں ہاتھ کے اصول کے مطابق ہو۔

\begin{align*}
\oint_C \mathbf{F} \cdot d\mathbf{r}
=
\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{n}\, dS
\tag{4}
\end{align*}

\begin{align*}
\text{Counterclockwise circulation} \quad = \quad \text{Curl integral}
\end{align*}

\vspace{1em}

اس سے یہ نتیجہ نکلتا ہے کہ اگر دو مختلف اورینٹڈ سطحیں $S_1$ اور $S_2$ ایک ہی boundary $C$ رکھتی ہوں، تو ان کے curl integrals برابر ہوں گے:

\begin{align*}
\iint_{S_1} (\nabla \times \mathbf{F}) \cdot \mathbf{n}_1\, dS
=
\iint_{S_2} (\nabla \times \mathbf{F}) \cdot \mathbf{n}_2\, dS
\end{align*}

یہ دونوں تکمل اس لیے برابر ہیں کیونکہ دونوں کا نتیجہ وہی boundary integral دیتا ہے، بشرطیکہ unit عمودی vectors صحیح orientation میں ہوں۔

\vspace{1em}

عام طور پر ہمیں کچھ ریاضیاتی شرائط درکار ہوتی ہیں تاکہ Stokes’ equation میں موجود تمام تکملات موجود ہوں۔ عام شرط یہ ہے کہ سمتیہ فیلڈ $\mathbf{F}$، منحنی $C$ اور سطح $S$ کے تمام فنکشنز اور ان کے مشتقات متصل (continuous) ہوں۔

\vspace{1em}

اگر $C$ xy-مستوی میں ایک بند منحنی ہو جو counterclockwise ہو اور $R$ وہ خطہ ہو جو $C$ سے گھرا ہوا ہو، تو:

\begin{align*}
dS = dx\,dy,\quad \mathbf{n} = \mathbf{k}
\end{align*}

اور:

\begin{align*}
(\nabla \times \mathbf{F}) \cdot \mathbf{n}
=
(\nabla \times \mathbf{F}) \cdot \mathbf{k}
=
\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}
\end{align*}

ان حالات میں Stokes’ equation درج ذیل صورت اختیار کر لیتی ہے:

\begin{align*}
\oint_C \mathbf{F} \cdot d\mathbf{r}
=
\iint_R
\left(
\frac{\partial N}{\partial x}
-
\frac{\partial M}{\partial y}
\right)
dx\,dy
\end{align*}

یہ Green’s Theorem کی circulation–curl form ہی ہے۔

اسی طرح الٹ عمل سے Green’s Theorem کو بھی del notation میں لکھا جا سکتا ہے:

\begin{align*}
\oint_C \mathbf{F} \cdot d\mathbf{r}
=
\iint_R (\nabla \times \mathbf{F}) \cdot \mathbf{k}\, dA
\tag{5}
\end{align*}

\begin{align*}

\end{align*}

\begin{align*}

=
\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{n}\, dS
\end{align*}

\frac{1}{\pi r^2}

\end{align*}

\frac{1}{\pi r^2}
\oint_C \mathbf{F}\cdot d\mathbf{r}
\tag{6}
\end{align*}

\begin{align*}
=
(7x - 6)(2 - 2x) + 2(2 - 2x)^2
\end{align*}

سادہ کرنے پر:

\begin{align*}
= -6x
\end{align*}

اب بیرونی تکمل:

\begin{align*}
\int_0^1 (-6x)\, dx
=
-3
\end{align*}

\vspace{1em}

\موٹا{حتمی جواب:}

\begin{align*}
\oint_C \mathbf{F}\cdot d\mathbf{r} = -3
\end{align*}

\begin{center}
{\Large کثیرسطحی (Polyhedral) سطحوں کے لیے Stokes’ Theorem کا ثبوت}
\end{center}

فرض کریں $S$ ایک کثیرسطحی سطح ہے جو محدود تعداد میں مسطح حصوں (مستوی regions) پر مشتمل ہے (شکل 16.66)۔

ہم ہر علیحدہ حصے پر Green’s Theorem لاگو کرتے ہیں۔ اس صورت میں دو قسم کے حصے ہوتے ہیں:

\begin{enumerate}
\item وہ حصے جو مکمل طور پر دوسرے حصوں سے گھرے ہوں
\item وہ حصے جن کی کم از کم ایک edge کسی دوسرے حصے سے متصل نہ ہو
\end{enumerate}

سطح $S$ کی boundary $\Delta$ ان edges پر مشتمل ہوتی ہے جو قسم (2) کے حصوں سے تعلق رکھتے ہیں۔

مثال کے طور پر شکل 16.66 میں مثلثی حصے $EAB$, $BCE$, اور $CDE$ سطح کا حصہ ہیں، جبکہ $ABCD$ boundary ہے۔

\vspace{1em}

ہر مثلث پر Green’s Theorem لگانے سے:

\begin{align*}
\sum \oint F\cdot dr
=
\sum \iint (\nabla \times \mathbf{F})\cdot \mathbf{n}\, dS
\tag{7}
\end{align*}

\vspace{1em}

بائیں طرف کے line integrals اندرونی edges پر آپس میں cancel ہو جاتے ہیں۔

مثال کے طور پر edge $BE$ پر ایک triangle میں integral ایک سمت میں ہے اور دوسرے میں اس کے الٹ، اس لیے وہ ختم ہو جاتے ہیں۔

اسی طرح $CE$ وغیرہ بھی cancel ہو جاتے ہیں۔

لہٰذا باقی صرف boundary رہ جاتی ہے:

\begin{align*}
\oint_{\partial S} \mathbf{F}\cdot d\mathbf{r}
=
\iint_S (\nabla \times \mathbf{F})\cdot \mathbf{n}\, dS
\end{align*}

یہی Stokes’ Theorem کا عمومی نتیجہ ہے۔

\section*{سوراخوں والی سطحوں کے لیے اسٹوکس کا قضیہ}

اسٹوکس کے قضیے کو ایسی اورینٹڈ سطح \( S \) تک بھی وسعت دی جا سکتی ہے جس میں ایک یا ایک سے زیادہ سوراخ ہوں۔ اس کی وضاحت گرین کے قضیے کی توسیع کے مشابہ ہے: سطح \( S \) پر سمتیہ فیلڈ \( \nabla \times \mathbf{F} \) کے نارمل جزو کا سطحی تکمل، سطح کی تمام حدی منحنیات (boundary curves) کے گرد خطی تکملات کے مجموعے کے برابر ہوتا ہے، بشرطیکہ ان منحنیات کو سطح کی دی گئی سمت کے مطابق طے کیا جائے۔

\bigskip

\موٹا{اہم شناخت}

ریاضی اور طبعی علوم میں درج ذیل شناخت بار بار سامنے آتی ہے:

\begin{align*}
\nabla \times (\nabla f) = 0
\quad \text{یا} \quad
\nabla \times \nabla f = 0
\tag{8}
\end{align*}

یہ شناخت ہر ایسے فعل \( f(x,y,z) \) کے لیے درست ہے جس کے دوسرے جزوی مشتقات مسلسل ہوں۔ اس کا ثبوت درج ذیل ہے:

\begin{align*}
\nabla \times \nabla f =
\begin{vmatrix}
\mathbf{i} & \mathbf{j} & \mathbf{k} \\
\frac{\partial}{\partial x} & \frac{\partial}{\partial y} & \frac{\partial}{\partial z} \\
\frac{\partial f}{\partial x} & \frac{\partial f}{\partial y} & \frac{\partial f}{\partial z}
\end{vmatrix}
\end{align*}

اس کو حل کرنے سے حاصل ہوتا ہے:

\begin{align*}
(\, f_{zy} - f_{yz} \,) \mathbf{i}
-
(\, f_{zx} - f_{xz} \,) \mathbf{j}
+
(\, f_{yx} - f_{xy} \,) \mathbf{k}
\end{align*}

اگر دوسرے جزوی مشتقات مسلسل ہوں تو مکسڈ مشتقات برابر ہوتے ہیں (نظریہ 2،  حصہ  14.3)، لہٰذا یہ سمتیہ صفر ہو جاتا ہے۔

\bigskip

\textbf{محافظ میدان اور اسٹوکس کا قضیہ}

 حصہ  16.3 میں ہم نے دیکھا کہ ایک سمتیہ فیلڈ \( \mathbf{F} \) کسی کھلے خطے \( D \) میں محافظ (conservative) اس وقت ہوتا ہے جب اس کا کسی بھی بند راستے کے گرد خطی تکمل صفر ہو۔ یہ بات مزید یہ ظاہر کرتی ہے کہ سادہ مربوط (simply connected) خطوں میں یہ شرط اس کے برابر ہے کہ

\begin{align*}
\nabla \times \mathbf{F} = 0
\end{align*}

\textbf{قضیہ 6: کرل صفر اور بند راستے کی خاصیت}

اگر کسی سادہ مربوط کھلے خطے \( D \) میں ہر نقطے پر \( \nabla \times \mathbf{F} = 0 \) ہو تو \( D \) میں کسی بھی ٹکڑوں میں ہموار بند منحنی \( C \) کے لیے:

\begin{align*}
\oint_C \mathbf{F} \cdot d\mathbf{r} = 0
\end{align*}

\bigskip

\textbf{ثبوت کا خاکہ}

قضیہ 6 کا ثبوت عموماً دو مراحل میں کیا جاتا ہے۔ پہلا مرحلہ سادہ بند منحنیات کے لیے ہوتا ہے۔ ٹوپولوجی (topology) کی ایک اہم شاخ کا ایک نظریہ یہ بیان کرتا ہے کہ ...



%===========================
%L91490

\bigskip

ہر قابلِ تفاضل (differentiable) سادہ بند منحنی \( C \)، جو کسی سادہ مربوط کھلے خطے \( D \) میں موجود ہو، ایک ہموار دو رُخا سطح \( S \) کی حد (boundary) ہوتی ہے جو خود بھی \( D \) کے اندر موجود ہوتی ہے۔ لہٰذا اسٹوکس کے قضیے کے مطابق:

\begin{align*}
\oint_C \mathbf{F} \cdot d\mathbf{r}
=
\iint_S (\nabla \times \mathbf{F}) \cdot \mathbf{n}\, ds
= 0
\end{align*}

\bigskip

دوسرا مرحلہ اُن منحنیات کے لیے ہے جو خود کو کاٹتی ہیں، جیسے کہ شکل 16.68 میں دکھائی گئی ہے۔ خیال یہ ہے کہ ایسی منحنیات کو سادہ حلقوں (simple loops) میں تقسیم کیا جائے، جن کے لیے قابلِ اورینٹیشن سطحیں موجود ہوں، پھر ہر حلقے پر الگ الگ اسٹوکس کے قضیے کو لاگو کیا جائے اور نتائج کو جمع کر لیا جائے۔

\bigskip

\textbf{شکل 16.68}

سادہ مربوط کھلے خطے میں ایسی قابلِ تفاضل منحنیات جو خود کو کاٹتی ہیں، انہیں ایسے حلقوں میں تقسیم کیا جا سکتا ہے جن پر اسٹوکس کا قضیہ لاگو کیا جا سکتا ہے۔

\bigskip

مندرجہ ذیل خاکہ سادہ مربوط کھلے خطوں میں متعین محافظ (conservative) میدانوں کے نتائج کو خلاصہ کرتا ہے:

\bigskip

\begin{align*}
\begin{array}{c}
\mathbf{F} \text{ conservative on } D
\end{array}
\;\;\Longleftrightarrow\;\;
\begin{array}{c}
\mathbf{F} = \nabla f \text{ on } D
\end{array}
\end{align*}

\bigskip

یہ تعلق درج ذیل نظریات اور نتائج کے ذریعے حاصل ہوتا ہے:

\begin{itemize}
\item نظریہ 1،  حصہ  16.3
\item نظریہ 2،  حصہ  13.3
\item سمتیہ شناخت: \( \nabla \times (\nabla f) = 0 \) (مساوات 8)
\end{itemize}

\bigskip

اسی سے درج ذیل ہم آہنگ نتائج حاصل ہوتے ہیں:

\begin{align*}
\oint_C \mathbf{F} \cdot d\mathbf{r} = 0
\quad \text{(کسی بھی بند راستے پر)}
\end{align*}

\begin{align*}
\nabla \times \mathbf{F} = 0 \quad \text{پورے خطے } D \text{ میں}
\end{align*}

یہ نتائج سادہ مربوطیت (simple connectivity) اور اسٹوکس کے قضیے کے ساتھ مل کر ایک دوسرے کے ہم معنی ہو جاتے ہیں۔

\subsection*{اسٹوکس کے قضیے کے ذریعے گردشی مقدار (Circulation) کا حساب}

درج ذیل مشقوں (1 تا 6) میں اسٹوکس کے قضیے کے سطحی تکمل (surface integral) کو استعمال کرتے ہوئے سمتیہ فیلڈ \( \mathbf{F} \) کی منحنی \( C \) کے گرد گردش (circulation) معلوم کریں، دی گئی سمت کے مطابق۔

\begin{enumerate}
\item
\begin{align*}
\mathbf{F} = x^2 \mathbf{i} + 2x \mathbf{j} + z^2 \mathbf{k}
\end{align*}
\( C \): بیضوی خطہ \( 4x^2 + y^2 = 4 \)، جو \( xy \)-سطح میں واقع ہے، اور اوپر سے دیکھنے پر مخالف گھڑی سمت (counterclockwise) میں ہے۔

\item
\begin{align*}
\mathbf{F} = 2y \mathbf{i} + 3x \mathbf{j} - z^2 \mathbf{k}
\end{align*}
\( C \): دائرہ \( x^2 + y^2 = 9 \)، جو \( xy \)-سطح میں واقع ہے، اور اوپر سے دیکھنے پر مخالف گھڑی سمت میں ہے۔

\item
\begin{align*}
\mathbf{F} = y \mathbf{i} + xz \mathbf{j} + x^2 \mathbf{k}
\end{align*}
\( C \): مثلث کی حد جو سطح \( x + y + z = 1 \) سے پہلی octant میں کٹی ہوئی ہے، اوپر سے دیکھنے پر مخالف گھڑی سمت میں۔

\item
\begin{align*}
\mathbf{F} = (y^2 + z^2)\mathbf{i} + (x^2 + z^2)\mathbf{j} + (x^2 + y^2)\mathbf{k}
\end{align*}
\( C \): مثلث کی حد جو سطح \( x + y + z = 1 \) سے پہلی octant میں کٹی ہوئی ہے، اوپر سے دیکھنے پر مخالف گھڑی سمت میں۔

\item
\begin{align*}
\mathbf{F} = (y^2 + z^2)\mathbf{i} + (x^2 + y^2)\mathbf{j} + (x^2 + y^2)\mathbf{k}
\end{align*}
\( C \): مربع جو خطوط \( x = \pm 1 \) اور \( y = \pm 1 \) سے محدود ہے، \( xy \)-سطح میں واقع ہے، اور اوپر سے دیکھنے پر مخالف گھڑی سمت میں۔

\item
\begin{align*}
\mathbf{F} = x^2 y^3 \mathbf{i} + \mathbf{j} + z \mathbf{k}
\end{align*}
\( C \): سلنڈر \( x^2 + y^2 = 4 \) اور نصف کرہ \( x^2 + y^2 + z^2 = 16, \; z \ge 0 \) کے تقاطع پر منحنی، اوپر سے دیکھنے پر مخالف گھڑی سمت میں۔
\end{enumerate}
