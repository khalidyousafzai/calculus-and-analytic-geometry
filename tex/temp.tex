\حصہ{مسئلہ سٹوکس  }\شناخت{حصہ_سمتی_میدان_مسئلہ_سٹوکس}
ہم نے  حصہ\حوالہ{حصہ_مستوی_میں_مسئلہ_گرین} میں دیکھا کہ دو بعدی میدان \عددی{\kvec{F} = M\ai + N\aj}  میں  نقطہ  \عددی{(x,y)}  پر کثافت  دائری بہاؤ یعنی    گردش  کو    غیر سمتی  مقدار  \عددی{\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}}  بیان کرتا ہے۔ تین  ابعاد  میں  مستوی میں نقطہ \عددی{P} کے گرد دائری بہاؤ   کو   سمتیہ کے ذریعے بیان کیا جاتا ہے۔یہ سمتیہ  دائری  بہاؤ  کے مستوی  کو عمودی ہو گا  (شکل\حوالہ{شکل_گردش_نقطہ_پر_بیان_سمتیہ})، اور سمتیہ  کا رخ دائری بہاؤ   کے خط    کے ساتھ دائیں ہاتھ کا تعلق رکھے گا۔سمتیہ کی لمبائی  سیال کے گھومنے کی شرح ظاہر کرتی ہے، جو  \عددی{P}  کے لحاظ سے دائری بہاؤ کا مستوی  جھکانے سے  عموماً  تبدیل ہو گی۔ہم کیا دیکھتے ہیں کہ  جب دائری بہاؤ کا سمتی رفتاری میدان \عددی{\kvec{F}=M\ai+N\aj+P\ak}  ہو،  زیادہ سے زیادہ دائری بہاؤ کا سمتیہ درج ذیل \اصطلاح{گردشی سمتیہ} \فرہنگ{گردشی سمتیہ}\حاشیہب{curl vector}\فرہنگ{curl vector} ہو گا۔

\begin{align}\label{مساوات_گردش_تعریف_الف}
\kvec{F}\text{گردش}=\left(\frac{\partial P}{\partial y} - \frac{\partial N}{\partial z}\right)\ai+
\left(\frac{\partial M}{\partial z} - \frac{\partial P}{\partial x}\right)\aj+
\left(\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}\right)\ak
\end{align}

یہ معلومات ہمیں\اصطلاح{مسئلہ سٹوکس }\فرہنگ{سٹوکس!مسئلہ}\حاشیہب{Stokes’ Theorem}\فرہنگ{Stokes!theorem}  دیتا ہے، جو مسئلہ گرین کی  دائری بہاؤ و گردشی روپ کی  تین بعدی   وسعت ہے۔

دھیان رہے کہ \عددی{(\kvec{F}\text{گردش}) \cdot \ak=\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}} اور \عددی{\kvec{F} = M(x,y)\ai + N(x,y)\aj}    کی صورت میں حصہ\حوالہ{حصہ_مستوی_میں_مسئلہ_گرین}   میں پیش کی گئی   تعریف   بلا تضاد  ہیں۔مساوات \حوالہ{مساوات_گردش_تعریف_الف}  میں گردش \عددی{\kvec{F}}   کا  کلیہ عموماً        درج ذیل علامتی عامل  کے ذریعے لکھا جاتا ہے۔
\begin{align}
\nabla =\ai\frac{\partial}{\partial x}+\aj\frac{\partial}{\partial y}+\ak\frac{\partial}{\partial z}
\end{align}

(علامت \عددی{\nabla} کو  ہم "ڈیل\حاشیہب{del} "  پڑھتے ہیں۔)  یہ عامل استعمال کر کے  گردش \عددی{\kvec{F}} \فرہنگ{گردش!کلیہ}\حاشیہب{curl}\فرہنگ{curl!equation} کو  ہم \عددی{\nabla\times\kvec{F}} لکھتے ہیں۔

\begin{align*}
\nabla \times \kvec{F}&=\begin{vmatrix}
\ai & \aj & \ak \\
\frac{\partial}{\partial x} &\frac{\partial}{\partial y} &\frac{\partial}{\partial z} \\
M & N & P
\end{vmatrix}\\
&=\left(\frac{\partial P}{\partial y} - \frac{\partial N}{\partial z}\right)\ai+
\left(\frac{\partial M}{\partial z} - \frac{\partial P}{\partial x}\right)\aj+
\left(\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}\right)\ak\\
&=\kvec{F}\text{گردش}
\end{align*}
مختصراً درج ذیل لکھا جائے گا۔
\begin{align}
  \kvec{F}\text{گردش}=\nabla \times \kvec{F}
\end{align}
%====================================================

\ابتدا{مثال}\موٹا{گردش \عددی{\kvec{F}} کی تلاش}\\
میدان \عددی{\kvec{F} = (x^2 - y)\ai + 4z\aj + x^2\ak}   کی گردش تلاش کریں۔

\موٹا{حل:}
\begin{align*}
\kvec{F}\text{گردش}&=\nabla\times \kvec{F}\\
&=
\begin{vmatrix}
\ai & \aj & \ak \\
\frac{\partial}{\partial x} &
\frac{\partial}{\partial y} &
\frac{\partial}{\partial z} \\
x^2 - y & 4z & x^2
\end{vmatrix}\\
&=\left(\frac{\partial}{\partial y}(x^2) - \frac{\partial }{\partial z}(4z)\right)\ai-
\left(\frac{\partial }{\partial x}(x^2) - \frac{\partial }{\partial z}(x^2-y)\right)\aj+
\left(\frac{\partial}{\partial x}(4z) - \frac{\partial}{\partial y}(x^2 - y)\right)\ak\\
&=(0 - 4)\ai-(2x - 0)\aj+(0 +1)\ak\\
&=-4\ai - 2x\aj + \ak
\end{align*}
\انتہا{مثال}


 ہم دیکھیں گے کہ عامل  \عددی{\nabla} کے دیگر استعمال بھی ہیں۔مثلاً ، غیر سمتی تفاعل \عددی{f(x,y,z)} پر  لگانے سے \اصطلاح{ڈھلوان }\فرہنگ{ڈھلوان!کلیہ}\حاشیہب{gradient}\فرہنگ{gradient!equation}  حاصل ہو گا۔
\begin{align*}
\nabla f=\frac{\partial f}{\partial x}\ai+\frac{\partial f}{\partial y}\aj+\frac{\partial f}{\partial z}\ak
\end{align*}
اسے ہم    \قول{ڈھلوان \عددی{f}} اور\حاشیہب{grad f} یا \قول{ڈیل \عددی{f}}کہیں \حاشیہب{del f} گے۔

\ابتدا{مسئلہ}\موٹا{مسئلہ سٹوکس}\\
    سمت بند سطح \عددی{S}   کی سرحد \عددی{C}   پر،      سطح کے عمودی اکائی سمتیہ \عددی{\kvec{n}} لے لحاظ سے خلاف گھڑی  چلتے ہوئے سمتیہ میدان  \عددی{\kvec{F}=M\ai+N\aj+P\ak} کا دائری بہاؤ   \عددی{S} پر \عددی{\nabla\times \kvec{F}\cdot\kvec{n}} کے تکمل کا  برابر ہو گا۔
\begin{align}
 \underbrace{\ointctrclockwise\limits_C \kvec{F} \cdot \dif \mathbf{r}}_{\text{\RL{خلاف گھڑی دائری بہاؤ}}}=\underbrace{\iint\limits_S (\nabla \times \kvec{F}) \cdot \an\, \dif \sigma}_{\text{\RL{تکمل گردش}}}  \tag{4}
\end{align}
\انتہا{مسئلہ}

ہم مساوات \حوالہء{4}  کو دیکھ کر کہہ سکتے ہیں کہ   ایسی دو مختلف سمت  بند سطحیں  \عددی{S_1} اور \عددی{S_2}  جن کی سرحد  \عددی{C} ایک ہو، کے تکمل گردش برابر ہوں گے۔

\begin{align*}
\iint_{S_1} (\nabla \times \kvec{F}) \cdot \an_1\, \dif \sigma=
\iint_{S_2} (\nabla \times \kvec{F}) \cdot \an_2\, \dif \sigma
\end{align*}


 دونوں تکمل  گردش  اس لئے ایک دوسرے کے   برابر ہوں گے  کیونکہ دونوں  مساوات \حوالہء{4} کے  بائیں ہاتھ  کے برابر ہیں، بشرطیکہ اکائی  عمودی  سمتیات \عددی{\kvec{n}_1} اور \عددی{\kvec{n}_2}   سطوح کی درست سمت بندی کریں۔

مسئلہ سٹوکس میں موجود  تکملات کی   وجودیت یقینی بنانے کے لئے ضرور ی ہے کہ \عددی{\kvec{F}}، \عددی{C}، اور \عددی{S} پر  ریاضیاتی شرائط مسلط ہوں گے۔ درکار عمومی شرائط  کہتے ہیں کہ تمام تفاعل، سمتیہ میدان، اور ان کے تفرق استمراری ہوں۔

اگر  \عددی{C} مستوی  \عددی{xy}  میں   موجود ، خلاف گھڑی سمت بند  منحنی   ہو، اور \عددی{R} مستوی \عددی{xy} میں ایک خطہ ہو جس کو \عددی{C} گھیرتی ہو، تب \عددی{\dif \sigma=\dif x\dif y}  اور درج ذیل ہو گا۔
\begin{align*}
(\nabla \times \kvec{F}) \cdot \an=(\nabla \times \kvec{F}) \cdot \ak=\left(
\frac{\partial N}{\partial x} - \frac{\partial M}{\partial y}\right)
\end{align*}
ان حالات میں مساوات سٹوکس درج ذیل صورت اختیار  کرتی ہے
\begin{align*}
\ointctrclockwise_C \kvec{F} \cdot \dif \mathbf{r}=\iint_R\left(\frac{\partial N}{\partial x}-\frac{\partial M}{\partial y}\right)\dif x\dif y
\end{align*}
جو مسئلہ گرین میں  موجود مساوات کی دائری بہاؤ و گردشی روپ ہے۔اسی طرح انہیں قدموں پر  الٹ  چلتے ہوئے  ہم مسئلہ گرین کے  دائری بہاؤ و گردشی  روپ کو دو بعدی میدان کے لئے  درج ذیل ڈیل علامتیت میں لکھ سکتے ہیں۔
\begin{align}
\ointctrclockwise_C \kvec{F} \cdot \dif \mathbf{r}=\iint_R (\nabla \times \kvec{F}) \cdot \ak\, \dif A \tag{5}
\end{align}
شکل\حوالہء{16.61} ملاحظہ کریں۔


\ابتدا{مثال}\quad \موٹا{نصف کرہ کے لئے مساوات سٹوکس کی تصدیق}\\
نصف کرہ \عددی{S:\, x^2+y^2+z^2=9,\, z\ge 0}، اس  کو گھیرتا دائرہ \عددی{C:\, x^2+y^2=9,\,z=0}، اور میدان \عددی{\kvec{F}=y\ai-x\aj} کے لئے مساوات \حوالہء{4} کی قیمت تلاش کریں۔
%=====================

\موٹا{حل:}\quad
مقدار معلوم روپ \عددی{\kvec{r}(\theta)=(3\cos\theta)\ai+(3\sin\theta)\aj,\,0\le \theta\le 2\pi} استعمال کرتے ہوئے اوپر  سے نظارہ کرتے ہوئے \عددی{C} کے گرد خلاف گھڑی دائری بہاؤ تلاش کرتے ہیں۔
\begin{align*}
\dif \mathbf{r}&=-3\sin \theta\,\dif \theta\,\ai+3\cos \theta\,\dif\theta\,\aj\\
\kvec{F} &= y\ai - x\aj=3\sin \theta\,\ai-3\cos \theta\,\aj\\
\kvec{F} \cdot \dif \mathbf{r}&=-9\sin^2 \theta\,\dif\theta-9\cos^2 \theta\,\dif\theta=-9\dif\theta\\
\ointctrclockwise_C \kvec{F} \cdot \dif \mathbf{r}&=\int_0^{2\pi} -9\,\dif \theta=-18\pi
\end{align*}

اب \عددی{\kvec{F}}   کے لئے تکمل گردش تلاش کرتے ہیں۔
\begin{align*}
\nabla \times \kvec{F}&= \left(\frac{\partial P}{\partial y}-\frac{\partial N}{\partial z}\right)\ai+
 \left(\frac{\partial M}{\partial z}-\frac{\partial P}{\partial x}\right)\aj+
 \left(\frac{\partial N}{\partial x}-\frac{\partial M}{\partial y}\right)\ak\\
&=(0-0)\ai+(0-0)\aj+(-1-1)\ak=-2\ak\\
\kvec{n}&=\frac{x\ai+y\aj+z\ak}{\sqrt{x^2+y^2+z^2}}=\frac{x\ai+y\aj+z\ak}{3}&&\text{\RL{بیرونی اکائی سمتیہ}}\\
\dif \sigma&=\frac{3}{z}\dif A&&\text{\RL{حصہ \حوالہء{16.5} کی مثال \حوالہء{5} میں \عددی{a=3}}}\\
\nabla\times \kvec{F}\cdot\kvec{n}\dif\sigma&=-\frac{2z}{3}\frac{3}{z}\dif A=-2\dif A
\end{align*}
یوں درج ذیل ہو گا۔
\begin{align*}
\iint_S (\nabla \times \kvec{F}) \cdot \an\, \dif\sigma=\iint_{x^2+y^2\le 9} -2\,\dif A=-18\pi
\end{align*}

اس طرح دائرے کی ہمراہ دائری بہاو  اور گردش کا تکمل نصف کرہ پر  برابر ہیں، جیسا کہ ہونا چاہیے تھا۔
\انتہا{مثال}

\ابتدا{مثال}\quad \موٹا{دائری بہاؤ کی تلاش}\\
مستوی \عددی{z=2} مخروط \عددی{z=\sqrt{x^2+y^2}} سے منحنی \عددی{C}  پر ملتا ہے۔ اس منحنی پر ، اوپر سے دیکھتے ہوئے خلاف گھڑی (شکل \حوالہء{16.62})، میدان \عددی{\kvec{F} = (x^2 - y)\ai + 4z\aj + x^2\ak}  کا ، دائری بہاؤ تلاش کریں۔

سطح: \quad
مسئلہ سٹوکس کے استعمال سے ہم دائری بہاؤ سطح پر تکمل سے حاصل کرتے ہیں۔ اوپر سے دیکھتے ہوئے \عددی{C} پر گھڑی خلاف چلنا   \ترچھا{اندرونی }عمود  \عددی{\kvec{n}}  لینے کے مترادف ہے  جس کا  \عددی{z}   جزو مثبت ہو گا۔

ہم مخروط کا  مقدار معلوم روپ لکھتے  ہیں۔

\begin{align*}
\mathbf{r}(r,\theta)=r\cos \theta\,\ai+r\sin \theta\,\aj+r\,\ak,\quad0 \le r \le 2,\; 0 \le \theta \le 2\pi
\end{align*}
اس طرح درج ذیل ہو گا۔
\begin{align*}
\kvec{n}&=\frac{\mathbf{r}_r \times \mathbf{r}_{\theta}}{|\mathbf{r}_r \times \mathbf{r}_{\theta}|}=
\frac{- r\cos \theta\,\ai-r\sin \theta\,\aj+r\,\ak}{r\sqrt{2}}&&
\text{\RL{حصہ \حوالہء{16.6} مثال \حوالہء{4}}}\\
&=\frac{-\cos \theta\,\ai - \sin \theta\,\aj + \ak}{\sqrt{2}}\\
\dif \sigma & = r\sqrt{2}\,\dif r\,\dif \theta&&\text{\RL{حصہ \حوالہء{16.6} مثال \حوالہء{4}}}\\
\nabla \times \kvec{F}&=-4\ai - 2x\aj + \ak&&\text{\RL{مثال \حوالہء{1}}}\\
&=-4\ai - 2r\cos \theta\,\aj + \ak&&  x=r\cos\theta
\end{align*}

\ointctrclockwise_C \kvec{F}\cdot \dif \mathbf{r}=\iint_S (\nabla \times \kvec{F}) \cdot \an\, \dif\delta
\end{align*}
 ہم  \عددی{\kvec{F}} کے دائرہ کار میں    اٹل  نقطہ \عددی{Q}، اور    اس نقطہ پر   مخصوص سمت \عددی{\mathbf{u}} منتخب کرتے  ہیں۔  فرض کریں  \عددی{C} ایک دائرہ ہے جس کا رداس \عددی{\rho} اور مرکز \عددی{Q} ہے اور جس کا مستوی \عددی{\kvec{u}} کو عمودی ہے۔اگر \عددی{Q} پر \عددی{\nabla\times\kvec{F}} استمراری ہو،   \عددی{C} میں محیط دائری قرص \عددی{S} پر \عددی{\nabla\times\kvec{F}} کے \عددی{\kvec{u}} ویں جزو کی اوسط قیمت \عددی{\rho\to 0} کی صورت میں \عددی{Q}  پر \عددی{\nabla\times\kvec{F}} کے \عددی{\kvec{u}} ویں جزو  کو پہنچے گی۔
\begin{align*}
(\nabla \times \kvec{F} \cdot \mathbf{u})_Q=\lim_{\rho\to 0}\frac{1}{\pi \rho^2}\iint_S (\nabla \times \kvec{F}) \cdot \mathbf{u}\, \dif\sigma
\end{align*}
آخری مساوات میں سطحی تکمل کی جگہ دائری بہاؤ لکھ کر درج ذیل حاصل ہو گا۔
\begin{align}
(\nabla \times \kvec{F} \cdot \mathbf{u})_Q=\lim_{\rho \to 0}\frac{1}{\pi \rho^2}\ointctrclockwise_C \kvec{F}\cdot \dif \mathbf{r}\tag{6}
\end{align}
مساوات \حوالہء{6}    کے بائیں ہاتھ کی قیمت اس صورت زیادہ سے زیادہ ہو گی جب \عددی{\kvec{u}} کا رخ \عددی{\mathbf{u}}، \عددی{\nabla \times \kvec{F}} کی سمت میں ہو۔چھوٹے \عددی{\rho} کے لئے مساوات \حوالہء{6}  کے دائیں ہاتھ   کا حد تخمیناً درج ذیل ہو گا، جو  \عددی{C} کے ہمراہ  دائری بہاؤ تقسیمِ رقبہ قرص    (دائری بہاؤ کی کثافت)ہے۔
\begin{align*}
\frac{1}{\pi \rho^2}\ointctrclockwise_C \kvec{F}\cdot \dif \mathbf{r}
\end{align*}
فرض کریں \عددی{Q} پر رداس \عددی{\rho} کی چپو چرخی رکھی جاتی ہے جس کا دھرا  \عددی{\kvec{u}}  کے ہمراہ ہے۔ سیال کا \عددی{C} کے ہمراہ دائری بہاؤ  چپو چرخی کے   گھومنے کی رفتار پر اثر انداز ہو گا۔ جب  دائری بہاؤ تکمل کی قیمت زیادہ سے زیادہ ہو اس صورت چپو چرخی تیز سے تیز گھومے گی؛ یوں  جب دھرا \عددی{\nabla\times\kvec{F}} کے رخ ہو، چپو چرخی تیز سے تیز گھومے گی (شکل \حوالہء{16.63})۔

\جزوحصہء{کثیر سطحی  سطح کے لئے مسئلہ سٹوکس کا ثبوت}
فرض کریں \عددی{S} کثیر سطحی سطح ہے جو  محدود تعداد  کے  مسطح حصوں  پر مشتمل  ہے (مثلاً شکل\حوالہء{16.66})۔ ہم  \عددی{S} کے علیحدہ علیحدہ حصوں  پر مسئلہ گرین کا اطلاق  کرتے ہیں۔  مستویات  کی دو  قسمیں ہیں۔
\begin{enumerate}[1.]
\item 
وہ  جو   ہر طرف سے  دوسرے حصوں سے گھیرا ہو۔
\item 
وہ جن کا ایک یا ایک سے زیادہ اطراف    کسی دوسرے حصے سے متصل نہ ہو۔
\end{enumerate}

سطح  \عددی{S}   کی سرحد \عددی{\Delta}    قسم   \عددی{2} مستویات کے ان کناروں پر مشتمل ہے جو دوسری مستویات سے متصل نہیں۔شکل \حوالہء{16.66} میں مثلثات   \عددی{EAB}، \عددی{BCE}، اور \عددی{CDE} سطح \عددی{S}  کے کچھ حصے کو ظاہر کرتے ہیں جبکہ \عددی{ABCD} سرحد \عددی{\Delta} کے کچھ حصے کو ظاہر کرتا ہے۔ تینوں مثلثات  پر بالترتیب مسئلہ گرین کا اطلاق کر کے نتائج جمع کر کے درج ذیل حاصل ہو گا۔
\begin{align}
\left(\ointctrclockwise_{EAB}+\ointctrclockwise_{BCE}+\ointctrclockwise_{CDE}\right) F\cdot \dif \kvec{r}=
\left( \iint _{EAB}+\iint_{BCE}+\iint_{CDE}\right)\nabla \times \kvec{F}\cdot \an\, \dif\sigma\tag{7}
\end{align}
چونکہ مساوات \حوالہء{7} میں  بائیں ہاتھ کے تین لکیری تکملات میں  اندرونی   (متصل  مثلثات کے مشترک)اضلاع  پر  لکیری تکملات  کے جوڑے ایک دوسرے کو کاٹتے ہیں لہٰذا  تینوں   مل کر  کنارہ \عددی{ABCDE} پر   واحد ایک  تکمل دیتے ہیں۔مثال کے طور پر  مثلث \عددی{ABE} کے ضلع  \عددی{BE} پر لکیری تکمل کی علامت \عددی{(\pm)} مثلث \عددی{EBC} کے اسی ضلع پر لکیری تکمل کی علامت  \عددی{(\mp)}کا الٹ ہو گی۔ضلع \عددی{CE} پر بھی ایسا ہو گا۔ یوں مساوات \حوالہء{7}   درج ذیل روپ اختیار کرتی ہے۔
\begin{align*}
\ointctrclockwise_{ABCDE} \kvec{F}\cdot \dif \mathbf{r}=\iint_{ABCDE}\nabla \times \kvec{F}\cdot \an\, \dif\sigma
\end{align*}
تمام اضلاع پر مسئلہ گرین کا اطلاق کر کے درج ذیل حاصل ہو گا جو  درحقیقت کثیر سطحی سطح \عددی{S} کا مسئلہ سٹوکس ہے۔ مزید عمومی  سطوح کے   لئے   ثبوت آپ کو   اعلٰی نصاب  میں  ملیں گے۔
\begin{align*}
\ointctrclockwise_{\Delta} \kvec{F}\cdot \dif\mathbf{r}=\iint_{S}\nabla \times \kvec{F}\cdot \an\, \dif\sigma

\جزوجزوحصہء{ایک اہم تماثل}
ریاضیات اور طبیعی سائنس میں درج ذیل تماثل بار بار پیش آتا ہے۔
\begin{align}
\nabla\times \nabla f=\kvec{0} \tag{8}
\end{align}
یہ تماثل  جو ہر اس تفاعل \عددی{f(x,y,z)} کے لئے  کارآمد ہے جس کے دوہرے جزوی تفرق استمراری ہوں، کہتا ہے کہ تفاعل \عددی{f(x,y,z)} کے ڈھلوان کی گردش   صفر ہو گی۔اس کا ثبوت درج ذیل ہے۔
\begin{align*}
\nabla \times \nabla f =
\begin{vmatrix}
\ai & \aj & \ak \\
\frac{\partial}{\partial x} & \frac{\partial}{\partial y} & \frac{\partial}{\partial z} \\
\frac{\partial f}{\partial x} & \frac{\partial f}{\partial y} & \frac{\partial f}{\partial z}
\end{vmatrix}
=(\, f_{zy} - f_{yz} \,) \ai-(\, f_{zx} - f_{xz} \,) \aj+(\, f_{yx} - f_{xy} \,) \ak
\end{align*}
ثانی  جزوی تفرق استمراری ہونے کی صورت میں  قوسین میں بند  مختلف اقسام کے ثانی  تفرق  برابر ہوں گے   (حصہ \حوالہء{14.3}، مسئلہ \حوالہء{2}) لہٰذا سمتیہ صفر ہو گا

\حصہء{بقائی میدان اور مسئلہ سٹوکس}
ہم نے حصہ \حوالہء{16.3} میں دیکھا  کہ کھلے خطہ \عددی{D} میں میدان \عددی{\kvec{F}} کا بقائی ہونا اس کے مترادف ہے کہ  \عددی{D} میں ہر بند راہ پر \عددی{\kvec{F}} کا لکیری تکمل صفر ہو۔یہ از خود، \ترچھا{سادہ تعلق }خطوں میں، اس کے مترادف ہے کہ \عددی{\nabla\times\kvec{F}=\kvec{0}} ہو۔

\ابتدا{مسئلہ}\quad \موٹا{\عددی{\nabla\times \kvec{F}=\kvec{0}}  اور بند حلقہ  خاصیت کا تعلق} \\
اگر خلا میں سادہ تعلق آزاد خطہ \عددی{D} کے ہر نقطہ پر \عددی{\nabla\times\kvec{F}=\kvec{0}} ہو تب \عددی{D} میں  ٹکڑوں میں  ہموار بند راہ \عددی{C} پر درج ذیل ہو گا۔
\begin{align*}
\ointctrclockwise_C \kvec{F}\cdot\dif\kvec{r}=0
\end{align*}

\موٹا{خاکہ}\quad
مسئلہ \حوالہء{6} کا ثبوت عموماً دو مراحل میں کیا جاتا ہے۔ پہلا مرحلہ سادہ بند منحنیات کے لیے ہو گا۔ اعلٰی ریاضیات کی ایک شاخ \اصطلاح{ربڑی ہندسہ}\فرہنگ{ربڑی ہندسہ}\حاشیہب{topology}\فرہنگ{topology}   کا ایک مسئلہ  کہتا ہے کہ سادہ تعلق کھلے خطہ \عددی{D} میں ہر قابل تفرق   سادہ بند منحنی  \عددی{C}  ایک ہموار دو طرفہ سطح \عددی{S}،   جو \عددی{D} میں پائی جاتی ہو، کی سرحد ہو گی ۔یوں مسئلہ سٹوکس کے تحت درج ذیل ہو گا۔
\begin{align*}
\ointctrclockwise_C \kvec{F} \cdot \dif\mathbf{r}=\iint_S (\nabla \times \kvec{F}) \cdot \an\, \dif\sigma= 0
\end{align*}
