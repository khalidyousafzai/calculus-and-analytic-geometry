\جزوحصہء{مسئلہ گرین استعمال کر کے رقبے کی تلاش}

اگر مستوی میں  سادہ منحنی \عددی{C}  اور  اس میں محیط خطہ \عددی{R} مسئلہ گرین   پر پورا اترتے ہوں، \عددی{R} کا رقبہ ذیل ہو گا۔
\begin{align}\label{مساوات_سمتی_میدان_مسئلہ_گرین_رقبہ}
&\text{\RL{رقبے کا مسئلہ گرین کلیہ}}\notag\\
&R\,\text{\RL{رقبہ}}\,=\frac{1}{2}\ointctrclockwise_C x\dif y-y\dif x
\end{align}
اس کی وجہ جانتے ہیں۔مساوات \حوالہ{مساوات_سمتی_تکمل_اخراج_اور_تکمل_پھیلاو}  کو الٹ چلا کر ذیل حاصل ہو گا۔
\begin{align*}
R\,\text{\RL{رقبہ}}\,&=\iint_R\dif y\dif x=\iint_R\left(\frac{1}{2}+\frac{1}{2}\right)\dif y\dif x\\
&=\ointctrclockwise_C \frac{1}{2}x\dif y-\frac{1}{2}y\dif x
\end{align*}
منحنیات میں محیط خطوں کے رقبے ، سوال \حوالہ{سوال_سمتی_میدان_رقبے_اکیس} تا  \حوالہ{سوال_سمتی_میدان_رقبے_چوبیس} میں، مساوات \حوالہ{مساوات_سمتی_میدان_مسئلہ_گرین_رقبہ} استعمال کر کے حاصل کریں۔

\ابتدا{سوال}\شناخت{سوال_سمتی_میدان_رقبے_اکیس}
دائرہ \عددی{\kvec{r}(t)=(a\cos t)\ai+(a\sin t)\aj,\, 0\le t\le 2\pi}
\انتہا{سوال}
\ابتدا{سوال}
ترخیم \عددی{\kvec{r}(t)=(a\cos t)\ai+(b\sin t)\aj,\, 0\le t\le 2\pi}
\انتہا{سوال}

\ابتدا{سوال}

\ابتدا{سوال}

\[\frac{1}{2}\ointctrclockwise_C x^2\dif y=-\ointctrclockwise_C xy\dif x=\frac{1}{3}\ointctrclockwise_C x^2\dif y-xy\dif x=A\bar{x}\]

\ابتدا{سوال}\ترچھا{مسئلہ گرین اور لاپلاس کی مساوات}\\

\[\frac{\partial^2 f}{\partial x^2}+\frac{\partial^2 f}{\partial y^2}=0\]
کو مطمئن کرتا ہو  تب ان تمام بند منحنیات \عددی{C}   پر جن پر مسئلہ گرین کا اطلاق ہوتا ہو درج ذیل ہو گا۔
\[\ointctrclockwise_C \frac{\partial f}{\partial y}\dif x-\frac{\partial f}{\partial x}\dif y=0\]
(اس کا الٹ بھی درست ہے: اگر لکیری تکمل ہر صورت صفر ہو، تب \عددی{f} لاپلاس کی مساوات کو   مطمئن کرے گا۔)

\[\kvec{F}=\left(\frac{1}{4}x^2y+\frac{1}{3}y^3\right)\ai+x\aj\]
(اشارہ: \عددی{\kvec{F}} کی گردش کہاں مثبت ہے؟)
