\حصہ{مقدار معلوم سطحیں}
ہم  مستوی میں منحنیات کو ذیل  تین مختلف طریقوں سے بیان  کر چکے ہیں۔
\begin{align*}
&y =f(x)&&\text{\RL{صریح روپ}}\\
&F(x,y)=0&&\text{\RL{خفی  روپ }}\\
&\kvec{r}(t)=f(t)\ai+g(t)\aj,\qquad  a\le t\le b&&\text{\RL{مقدار معلوم سمتیہ روپ}}
\end{align*}
فضا میں سطوح کی بھی ایسی تعریفیں موجود ہیں۔
\begin{align*}
&z=f(x,y)&&\text{\RL{صریح  روپ}}\\
&F(x,y,z)=0&&\text{\RL{خفی  روپ}}
\end{align*}
ایک مقدار معلوم روپ  بھی پایا جاتا  ہے جس میں سطح پر نقطے کا مقام دو متغیرات کا سمتی تفاعل  دیتا ہے۔ اس حصے میں سطحی رقبے اور سطحی تکملات کے مطالعہ کو ان سطحوں تک  وسعت دی جائے گی جنہیں مقدار معلوم روپ  میں بیان کیا گیا ہو۔

\جزوحصہء{سطحوں کی مقدار معلوم نمائندگی}
فرض کریں  استمراری  سمتی تفاعل جو   مستوی \عددی{uv}  کے  خطہ \عددی{R} پر معین اور \عددی{R} کے  اندرونی  حصے میں ایک ایک مطابقت رکھتا ہے  (شکل \حوالہ{شکل_سمتی_میدان_سطح_مقدار_معلوم_روپ}) کو درج ذیل  مساوات ظاہر کرتی ہے۔
\begin{align}\label{مساوات_مقدار_معلوم_سطح_الف}
\kvec{r}(u,v)=f(u,v)\ai+g(u,v)\aj+h(u,v)\ak
\end{align}

ہم مثال \حوالہء{2}کی مقدار معلوم نمائندگی استعمال کرتے ہیں:

\begin{align*}
\kvec{r}(\phi,\theta)=&(a\sin\phi\cos\theta)\ai+(a\sin\phi\sin\theta)\aj+(a\cos\phi)\ak\\
&0\le \phi \le \pi,\qquad0\le \theta \le 2\pi
\end{align*}




اب \عددی{\kvec{r}_\phi \times \kvec{r}_\theta} نکالتے ہیں۔

\begin{align*}
\kvec{r}_\phi \times \kvec{r}_\theta&=
\begin{vmatrix}
\ai & \aj & \ak \\
a\cos\phi\cos\theta & a\cos\phi\sin\theta & -a\sin\phi \\
-a\sin\phi\sin\theta & a\sin\phi\cos\theta & 0
\end{vmatrix}\\
&=(a^2\sin^2\phi\cos\theta)\ai+(a^2\sin^2\phi\sin\theta)\aj+(a^2\sin\phi\cos\phi)\ak
\end{align*}

یوں درج ذیل ہو گا۔
\begin{align*}
\left|\kvec{r}_\phi \times \kvec{r}_\theta\right|&=\sqrt{a^4\sin^4\phi\cos^2\theta+a^4\sin^4\phi\sin^2\theta+a^4\sin^2\phi\cos^2\phi}\\
&=\sqrt{a^4\sin^4\phi+a^2\sin^2\phi\cos^2\phi}=\sqrt{a^4\sin^2\phi(\sin^2\phi+\cos^2\phi)}\\
&=a^2\sqrt{\sin^2\phi}=a^2\sin\phi.
\end{align*}
چونکہ \عددی{0\le \phi \le \pi} پر \عددی{\sin\phi \ge 0} ہو گا لہٰذا سطحی رقبہ درج ذیل ہو گا۔
\begin{align*}
A&=\int_{0}^{2\pi}\int_{0}^{\pi}a^2\sin\phi\,\dif \phi\,\dif\theta\\
&\int_{0}^{2\pi}\left[-a^2\cos\phi\right]_{0}^{\pi}\,\dif\theta=\int_{0}^{2\pi}2a^2\,\dif \theta=4\pi a^2\quad \text{\RL{مربع اکائیاں}}
\end{align*}
یہ نتیجہ کرہ کے معلوم کلیہ  کے مطابق ہے۔
\انتہا{مثال}

\ابتدا{تعریف}\quad  سطحی تکمل برائے مقدار معلوم  سطح

اگر سطح  \عددی{S}  استمراری ہو  جس  کا مقدار معلوم روپ
\begin{align*}
\kvec{r}(u,v) =& f(u,v)\ai + g(u,v)\aj + h(u,v)\ak\\
&a \le u \le b,\; c \le v \le d
\end{align*}
ہو، اور  \عددی{G(x,y,z)}  ایک استمراری تفاعل ہو جو  \عددی{S} پر  معین  ہو، تب  \عددی{S} پر \عددی{G} کا تکمل درج ذیل ہو گا۔
\begin{align*}
\iint_S G(x,y,z)\, \dif \sigma=\int_c^d \int_a^bG\big(f(u,v), g(u,v), h(u,v)\big)\,\left| \kvec{r}_u \times \kvec{r}_v\right| \, \dif u\, \dif v
\end{align*}
\انتہا{تعریف}

\ابتدا{مثال}\quad مقدار معلوم روپ کی  سطح پر تکمل

تفاعل \عددی{G(x,y,z) = x^2} کا تکمل مخروطی سطح \عددی{z = \sqrt{x^2 + y^2},\, 0 \le z \le 1} پر معلوم کریں۔

حل:\quad
مثال \حوالہء{1} اور  مثال{4} کا  کام آگے بڑھاتے ہوئے \عددی{|\kvec{r}_r\times \kvec{r}_{\theta}|=\sqrt{2}r} لے کر درج ذیل ہو گا۔
\begin{align*}
\iint_S x^2 \, \dif \sigma&=\int_0^{2\pi} \int_0^1(r^2 \cos^2 u)(\sqrt{2}\, r)\, \dif r\, \dif u&& x=r\cos\theta\\
&=\sqrt{2}\int_0^{2\pi} \int_0^1r^3 \cos^2 \theta \, \dif r\, \dif \theta\\
&=\frac{\sqrt{2}}{4}\int_0^{2\pi} \cos^2 \theta \, \dif \theta=\frac{\sqrt{2}}{4}\left[  \frac{\theta}{2}+\frac{1}{4}\sin 2\theta \right]_0^{2\pi}=\frac{\pi\sqrt{2}}{2}
\end{align*}

سطح پر  \عددی{x=x}، \عددی{y=x^2}، اور \عددی{z=z} ہیں لہٰذا مقدار معلوم روپ \عددی{
\kvec{r}(x,z)=x\ai+x^2\aj+z\ak
} ہو گا جہاں \عددی{
{0\le x\le1},\,{0\le z\le 4}
}ہے۔

مماسی سمتیات کا صلیبی ضرب درج ذیل ہو گا۔

\begin{align*}
\kvec{r}_x \times \kvec{r}_z =
\begin{vmatrix}
\ai & \aj & \ak \\
1 & 2x & 0 \\
0 & 0 & 1
\end{vmatrix}
= 2x\,\ai - \aj
\end{align*}

سطح پر باہر رخ عمودی سمتیہ درج ذیل ہے۔

\begin{align*}
\mathbf{n}=\frac{\kvec{r}_x\times\kvec{r}_z}{|\kvec{r}_x\times\kvec{r}_z|}=
\frac{2x\,\ai - \aj}{\sqrt{4x^2 + 1}}
\end{align*}

سطح پر  \عددی{y=x^2} ہے لہٰذا اس پر سمتیہ میدان درج ذیل ہو گا۔
\begin{align*}
\mathbf{F} = yz\ai+x\aj-z^2\ak=x^2 z\,\ai + x\,\aj - z^2\,\ak
\end{align*}
یوں 
\begin{align*}
\mathbf{F} \cdot \mathbf{n}&=\frac{1}{\sqrt{4x^2+1}}((x^2z)(2x)+(x)(-1)+(-z^2)(0))\\
&=\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\end{align*}

اور \عددی{\kvec{F}} کا  سطح  سے  باہر رخ گزرتا  بہاؤ درج ذیل ہو گا۔
\begin{align*}
\iint_S \mathbf{F} \cdot \mathbf{n}\, \dif \sigma &=\int_0^4 \int_0^1\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\left| \kvec{r}_x \times \kvec{r}_z \right| \dif x\, \dif z\\
&=\int_0^4 \int_0^1\frac{2x^3 z - x}{\sqrt{4x^2 + 1}}
\sqrt{4x^2+1}\dif x\, \dif z\\
&=\int_0^4 \int_0^1 (2x^3 z - x)\, \dif x\, \dif z=\int_0^4\left[ \frac{x^4 z}{2} - \frac{x^2}{2} \right]_0^1 \dif z\\
&\int_0^4 \frac{z - 1}{2}\, \dif z=\left.\frac{1}{4} (z - 1)^2\right]_0^4\\
&=\frac{1}{4}(9)-\frac{1}{4}(1)=2
\end{align*}

محور \عددی{z} کے گرد  سطح تشاکل ہے لہٰذا \عددی{\bar{x}=0} اور \عددی{\bar{y}=0}  ہو گا۔ ہمیں \عددی{\bar{z} = \frac{M_{xy}}{M}} تلاش کرنی ہے۔ مثال \حوالہء{1} اور مثال \حوالہء{4} کی طرح چلتے ہوئے 
\begin{align*}
\kvec{r}(r,\theta) =r \cos \theta\, \ai+r \sin \theta\, \aj+r\, \ak,\quad1 \le r \le 2,\; 0 \le \theta \le 2\pi
\end{align*}
اور
\begin{align*}
\left| \kvec{r}_r \times \kvec{r}_{\theta} \right| = \sqrt{2}\, r
\end{align*}
ہو گا۔ یوں درج ذیل ہو گا۔

\begin{align*}
M &=
\iint_S \delta \, \dif \sigma=\int_0^{2\pi} \int_1^2 \delta  \sqrt{2}\, r\, \dif r\, \dif \theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\left[ \frac{r^2}{2} \right]_1^2 \dif\theta=\delta\sqrt{2}\int_0^{2\pi}\left(2-\frac{1}{2}\right)\dif\theta\\
&=\delta \sqrt{2}\left[\frac{3\theta}{2}\right]_0^{2\pi}=3\pi\delta\sqrt{2}
\end{align*}
%
\begin{align*}
M_{xy}&=\iint_S \delta z\, \dif \sigma=\int_0^{2\pi} \int_1^2\delta r \sqrt{2}\, r\, \dif r\, \dif\theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\int_1^2 r^2\dif r\dif\theta =\delta\sqrt{2}\int_0^{2\pi}\left[\frac{r^3}{3}\right]_1^2\dif\theta\\
&=\delta\sqrt{2}\int_0^{2\pi}\frac{7}{3}\dif\theta=\frac{14}{3}\pi\delta\sqrt{2}
\end{align*}
%
\begin{align*}
\bar{z}=\frac{M_{xy}}{M}=\frac{14\pi\delta\sqrt{2}}{3(3\pi\delta\sqrt{2})}=\frac{14}{9}
\end{align*}
خول کا مرکز کمیت نقطہ \عددی{(0,0,\tfrac{14}{9})} پر پایا جاتا ہے۔

\جزوحصہء{مقدار معلوم روپ کی مزید مثالیں}

%53
\ابتدا{سوال}
\begin{enumerate}
\item[(الف)]
  محور \عددی{z}  کے گرد    مستوی  \عددی{xz}    میں  دائرہ \عددی{C}  کو خلا میں  گھمانے سے اندرسہ حاصل ہو گا۔ (دکھائے گئے شکل سے رجوع کریں۔)  اگر \عددی{C} کا  رداس \عددی{r>0} اور مرکز \عددی{(R,0,0)} ہو ثابت کریں کہ اندرسہ کا مقدار معلوم روپ درج ذیل ہے
\[
\mathbf r(u,v)
=(R+r\cos u)\cos v\,\mathbf i
+(R+r\cos u)\sin v\,\mathbf j
+r\sin u\,\mathbf k
\]
جہاں \عددی{0\le u\le2\pi} اور \عددی{0\le v\le2\pi} ہیں۔

\item[(ب)]
ثابت کریں کہ اندرسہ کا  رقبہ \عددی{A=4\pi^2Rr} ہے۔
\end{enumerate}
\انتہا{سوال}
%54
\ابتدا{سوال}\موٹا{سطح طواف کا مقدار معلوم روپ}\quad
فرض کریں مقدار معلوم شدہ منحنی \عددی{C:(f(u),g(u))}  محور  \عددی{x} کے گرد گھمائی جاتی ہے، جہاں   \عددی{a\le u\le b} پر \عددی{g(u)>0} ہے۔

\begin{enumerate}
\item[(الف)]
 دکھائیں کہ سطح طواف کا مقدار معلوم روپ درج ذیل ہو گا
\[
\mathbf r(u,v)=f(u)\mathbf i+g(u)\cos v\,\mathbf j+g(u)\sin v\,\mathbf k
\]
جہاں  مستوی \عددی{xy} سے نقطہ \عددی{\kvec{r}(u,v)}   تک  زاویہ  \عددی{0\le v\le 2\pi}  ہے ۔ (دی گئی شکل ملاحظہ کریں۔)  دھیان  دیں کہ \عددی{f(u)} محور  طواف  کے  ہمراہ فاصلے دیتا ہے اور \عددی{g(u)} محور طواف سے فاصلے دیتا ہے۔

\item[(ب)] 
اس سطح کا  مقدار معلوم روپ  معلوم کریں جو منحنی \عددی{x=y^2}, \عددی{y\ge0} کو محور  \عددی{x} کے گرد گھمانے سے حاصل ہوگا۔
\end{enumerate}
\انتہا{سوال}
%55
\ابتدا{سوال}\موٹا{ترخیمی سطح کا مقدار معلوم روپ}\quad
\begin{enumerate}
\item[(الف)]
ترخیم \عددی{\tfrac{x^2}{a^2}+\tfrac{y^2}{b^2}=1}      (حصہ \حوالہء{3.5} میں مثال \حوالہء{13})کا مقدار معلوم روپ درج ذیل ہے۔
\عددی{x=a\cos \theta,\; y=b\sin \theta,\; 0\le \theta\le2\pi}
کروی محدد کے زاویات \عددی{\theta} اور \عددی{\phi} استعمال کرتے ہوئے ثابت کریں
 ترخیم  \عددی{\tfrac{x^2}{a^2}+\tfrac{y^2}{b^2}+\tfrac{z^2}{c^2}=1}  کا مقدار معلوم روپ درج ذیل ہے۔
\[
\mathbf r(u,\phi)=a\cos u\cos\phi\,\mathbf i+b\sin u\cos\phi\,\mathbf j+c\sin\phi\,\mathbf k
\]
\item[(ب)]
ترخیمی سطح  کے رقبے کا  تکمل لکھیں، لیکن اسے حل نہ کریں۔
\end{enumerate}
\انتہا{سوال}
%56
\ابتدا{سوال}\موٹا{یک  ورقی  قطع زائد سطح}
\begin{enumerate}
\item[(الف)] 
یک ورقی قطع زائد سطح 
\عددی{
x^2+y^2-z^2=1
}
کا مقدار معلوم روپ  دائرہ \عددی{x^2+y^2=r^2}  سے وابسطہ  زاویہ \عددی{\theta} اور  ہذلولی تفاعل  \عددی{r^2-z^2=1}   سے وابسطہ ہذلولی مقدار معلوم  \عددی{u}  استعمال کر کے حاصل  کریں۔ (حصہ \حوالہء{7.8} میں سوال \حوالہء{84} سے رجوع کریں۔)
\item[(ب)] 
جزو الف کے نتیجہ کو   درج ذیل کے لئے عمومی بنائیں۔
\[
\frac{x^2}{a^2}+\frac{y^2}{b^2}-\frac{z^2}{c^2}=1
\]
\end{enumerate}
\انتہا{سوال}
