\جزوحصہء{سطحی رقبہ}

\ابتدا{سوال}
مکافی جسم \عددی{x^2+y^2-z=0} سے مستوی \عددی{z=2}  جو  سطح  کاٹتی ہے اس  کا رقبہ تلاش کریں۔
\انتہا{سوال}
\ابتدا{سوال}
مکافی جسم \عددی{x^2+y^2-z=0} سے مستویات \عددی{z=2} اور \عددی{z=6}   جو   پٹی کاٹتی ہے اس  کا رقبہ تلاش کریں۔
\انتہا{سوال}
\ابتدا{سوال}
مستوی \عددی{x+2y+2z=5} سے ایک بیلن ، جس کی دیواریں \عددی{x=y^2} اور \عددی{x=2-y^2} ہیں،  جو خطہ کاٹتا ہے اس کا رقبہ تلاش کریں۔
\انتہا{سوال}
\ابتدا{سوال}
سطح \عددی{x^2-2z=0} کے اس حصے کا رقبہ تلاش کریں جو مستوی \عددی{xy}  میں واقع  اس مثلث  کے اوپر پایا جاتا ہے جس کو لکیر \عددی{x=\sqrt{3}}، \عددی{y=0}، اور \عددی{y=x} گھیرتی ہیں۔
\انتہا{سوال}
\ابتدا{سوال}
سطح \عددی{x^2-2y-2z=0} کے اس حصے کا رقبہ تلاش کریں جو مستوی \عددی{xy}  میں واقع  اس مثلث  کے اوپر پایا جاتا ہے جس کو لکیر \عددی{x=2}، \عددی{y=0}، اور \عددی{y=3x} گھیرتی ہیں۔
\انتہا{سوال}
\ابتدا{سوال}
کرہ \عددی{x^2+y^2+z^2=2} سے مخروط \عددی{z=\sqrt{x^2+y^2}} جو ٹوپی کاٹتا ہے اس کا رقبہ تلاش کریں۔
\انتہا{سوال}
%Q7
\ابتدا{سوال}
مستوی \عددی{z=cx} سے  بیلن  \عددی{x^2+y^2=1} جو  ترخیم کاٹتا ہے اس کا رقبہ تلاش کریں۔
\انتہا{سوال}
\ابتدا{سوال}
بیلن \عددی{x^2+z^2=1} کے اس بالائی حصے کا رقبہ تلاش کریں جو  مستویات \عددی{x=\pm 1\!/\!2} اور \عددی{y=\pm 1\!/\!2} کے بیچ پایا جاتا ہے۔
\انتہا{سوال}
\ابتدا{سوال}
مستوی \عددی{yz} میں واقع    چھلا \عددی{1\le y^2+z^2\le 4}  کے اوپر مکافی جسم \عددی{x=4-y^2-z^2}  کا جو حصہ پایا جاتا ہے اس کا رقبہ تلاش کریں۔
\انتہا{سوال}
\ابتدا{سوال}
مکافی سطح \عددی{x^2+y+z^2=2} سے مستوی \عددی{y=0} جو سطح کاٹتی ہے اس کا رقبہ تلاش کریں۔
\انتہا{سوال}
\ابتدا{سوال}
مستوی \عددی{xy} میں واقع مربع \عددی{R:\,1\le x\le 2,\, 0\le y\le 1} کے اوپر سطح \عددی{x^2-2\ln x+\sqrt{15}-z=0} کا جو حصہ پایا جاتا ہے اس کا رقبہ تلاش کریں۔
\انتہا{سوال}
\ابتدا{سوال}
مستوی \عددی{xy} میں واقع مربع \عددی{R:\,0\le x\le 1,\, 0\le y\le 1} کے اوپر سطح \عددی{2x^{3/2}+2y^{3/2}-3z=0}    کا جو حصہ پایا جاتا ہے اس کا رقبہ تلاش کریں۔
\انتہا{سوال}

\جزوحصہء{سطحی تکملات}
\ابتدا{سوال}
ثمن اول سے مستویات \عددی{x=a}، \عددی{y=a}، اور \عددی{z=a} جو مکعب کاٹتی ہیں اس کی سطح پر  \عددی{g(x,y,z)=x+y+z} کا تکمل حاصل کریں۔
\انتہا{سوال}
\ابتدا{سوال}
ثمن اول سے  محددی  مستویات اور  مستویات  \عددی{x=2} اور \عددی{y+z=1}  جو پچر کاٹتی ہیں اس کی سطح پر \عددی{g(x,y,z)=y+z} کا تکمل حاصل کریں۔
\انتہا{سوال}
\ابتدا{سوال}
ثمن اول سے مستویات \عددی{x=a}، \عددی{y=b}، اور \عددی{z=c} جو  مستطیلی  ٹھوس جسم کاٹتی ہیں اس کی سطح پر \عددی{g(x,y,z)=xyz} کا تکمل لیں۔
\انتہا{سوال}
%Q16
\ابتدا{سوال}
 مستویات \عددی{x=\pm a}، \عددی{y=\pm b}، اور \عددی{z=\pm c}  جو مستطیلی  ٹھوس جسم گھیرتی ہیں اس کی سطح پر \عددی{g(x,y,z)=xyz} کا تکمل لیں۔
\انتہا{سوال}
\ابتدا{سوال}
مستویات \عددی{x=0}، \عددی{x=1}، اور \عددی{z=0}  مکافی   بیلن \عددی{y^2+4z=16} سے جو سطح کاٹتی ہیں اس پر  \عددی{g(x,y,z)=x\sqrt{y^2+4}} کا تکمل لیں۔
\انتہا{سوال}

\جزوحصہء{سطح سے گزرتا بہاؤ}
دی گئی سطح کے  ٹکڑے   سے  میدان \عددی{\kvec{F}} کا گزرتا  بہاؤ سوال \حوالہ{سوال_سمتی_میدان_سطح_بہا_الف} اور
  سوال \عددی{سوال_سمتی_میدان_سطح_بہا_ب} میں تلاش کریں۔

\ابتدا{سوال}\شناخت{سوال_سمتی_میدان_سطح_بہا_الف}
\(\kvec{F}(x,y,z)=-\ai+2\aj+3\ak\)\\
مستطیل سطح \عددی{z=0,\, 0\le x\le 2,\, 0\le y\le 3} جس کا رخ \عددی{\ak} ہے سطح \عددی{S} ہے۔
\انتہا{سوال}
\ابتدا{سوال}\شناخت{سوال_سمتی_میدان_سطح_بہا_ب}
\(\kvec{F}(x,y,z)=yx^2\ai-2\aj+xz\ak\)\\
مستطیل سطح \عددی{y=0,\, -1\le x\le 2,\, 2\le z\le 7} جس کا رخ \عددی{-\aj} ہے سطح \عددی{S} ہے۔
\انتہا{سوال}

ثمن اول میں کرہ \عددی{x^2+y^2+z^2=a^2} کے   ٹکڑا  سے، جس کا رخ مبدا سے دوری  جانب  ہے، میدان \عددی{\kvec{F}} کا  گزرتا 
بہاؤ سوال \حوالہ{سوال_سمتی_میدان_کرہ_بہا_الف} تا \حوالہ{سوال_سمتی_میدان_کرہ_بہا_ب} میں تلاش کریں۔

\ابتدا{سوال}\شناخت{سوال_سمتی_میدان_کرہ_بہا_الف}
\(\kvec{F}(x,y,z)=z\ak\)
\انتہا{سوال}
\ابتدا{سوال}
\(\kvec{F}(x,y,z)=-y\ai+x\aj\)
\انتہا{سوال}
\ابتدا{سوال}
\(\kvec{F}(x,y,z)=y\ai-x\aj+\ak\)
\انتہا{سوال}
\ابتدا{سوال}
\(\kvec{F}(x,y,z)=y\ai-x\aj+\ak\)
\انتہا{سوال}
\ابتدا{سوال}
\(\kvec{F}(x,y,z)=zx\ai+zy\aj+z^2\ak\)
\انتہا{سوال}
\ابتدا{سوال}
\(\kvec{F}(x,y,z)=x\ai+y\aj+z\ak\)
\انتہا{سوال}
\ابتدا{سوال}\شناخت{سوال_سمتی_میدان_کرہ_بہا_ب}
\(\kvec{F}(x,y,z)=\tfrac{x\ai+y\aj+z\ak}{\sqrt{x^2+y^2+z^2}}\)
\انتہا{سوال}
\ابتدا{سوال}
قطع مکافی  بیلن \عددی{z=4-y^2} سے مستویات \عددی{x=0}، \عددی{x=1}، اور \عددی{z=0}   جو سطح  کاٹتی ہیں اس سے اوپر رخ گزرتا بہاؤ میدان \عددی{\kvec{F}(x,y,z)=z^2\ai+x\aj-3z\ak} کے لئے تلاش کریں۔
\انتہا{سوال}

\ابتدا{سوال}
مکافی سطح \عددی{z=x^2+y^2}  کے نچلے حصے سے مستوی \عددی{z=1}  جو  سطح  کاٹتی ہے اس سے  باہر رخ (محور \عددی{z} سے دور جانب)  گزرتا بہاؤ میدان \عددی{\kvec{F}(x,y,z)=4x\ai+4y\aj+2\ak} کے لئے  تلاش کریں۔

\انتہا{سوال}

بیلن \عددی{y^2+z^2=9,\, z\ge 0} سے مستویات \عددی{x=0} اور \عددی{x=3} جو سطح کاٹتی ہیں  (یہ چطح مثال\حوالہ{مثال_سمتی_میدان_بیلنی_چادر} کی سطح سے مشابہت رکھتی ہے)اس کا   وسطانی  مرکز تلاش کریں۔
\انتہا{سوال}
\ابتدا{سوال}\شناخت{سوال_سمتی_میدان_بیلن_مخروط}
